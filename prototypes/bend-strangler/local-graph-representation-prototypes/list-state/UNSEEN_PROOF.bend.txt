import Base

import ../../../../vendor/bendlib/packages/bend-mathlib/nat.bend as NatFacts

law unseen_insertion_nonincreasing:
  for +catalog: Own.Dictionary<Maybe<&2, Core.Declaration>>
  for +visited: Own.Dictionary<Bool>
  for +identity: Nat
  {
    Nat.is_le(
      Rank.unseen_catalog(Own.set(Bool, visited, identity, True{}), catalog),
      Rank.unseen_catalog(visited, catalog)
    )
    == True{}
    : Bool
  }

# Every successful selected root is excluded from unseen weight after insertion.
# No absent-root-identity assumption: prior visited roots still satisfy this bound.

def Support.weight_of_union_nonincreasing(seen, inserted, amount):
  match seen inserted:
    case True{} True{}: {==}
    case True{} False{}: {==}
    case False{} True{}: NatFacts.zero_le(amount)
    case False{} False{}: NatFacts.le_refl(amount)

def Support.unseen_declaration_nonincreasing(visited, identity, declaration):
  match declaration:
    case None{}: {==}
    case Some{Core.Declaration{_, +entry_identity, _, _, +references}}:
      exact = Membership.insertion_membership_exact(Bool, visited, identity, entry_identity, True{})
      %Equal.sym(
        Bool,
        Own.contains(Bool, Own.set(Bool, visited, identity, True{}), entry_identity),
        Bool.or(Nat.is_eq(entry_identity, identity), Own.contains(Bool, visited, entry_identity)),
        exact
      ):
        {
          Nat.is_le(
            Support.weight(_, List.length(&2, Core.Reference, references)),
            Support.weight(
              Own.contains(Bool, visited, entry_identity),
              List.length(&2, Core.Reference, references)
            )
          )
          == True{}
          : Bool
        }
      Support.weight_of_union_nonincreasing(
        Own.contains(Bool, visited, entry_identity),
        Nat.is_eq(entry_identity, identity),
        List.length(&2, Core.Reference, references)
      )

def unseen_insertion_nonincreasing(catalog, visited, identity):
  match catalog:
    case Own.DNil{}: {==}
    case Own.DCon{_, +declaration, +rest, _}:
      head = Support.unseen_declaration_nonincreasing(visited, identity, declaration)
      suffix = unseen_insertion_nonincreasing(rest, visited, identity)
      NatFacts.add_le_add(
        Rank.unseen_declaration(Own.set(Bool, visited, identity, True{}), declaration),
        Rank.unseen_catalog(Own.set(Bool, visited, identity, True{}), rest),
        Rank.unseen_declaration(visited, declaration),
        Rank.unseen_catalog(visited, rest), head, suffix)
