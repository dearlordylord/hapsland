import Base
import ./Own.bend as Own
import ./INVARIANT.bend as Invariant
import ../../../../vendor/bendlib/packages/bend-mathlib/nat.bend as NatFacts
import ../../../../vendor/bendlib/packages/bend-mathlib/bool.bend as BoolFacts

# Exact approved statement, isolated from the other five still-open laws.
law cached_size_exact:
  for -V: Data
  for +catalog: Own.Dictionary<V>
  for valid: {Invariant.valid(V, catalog) == True{} : Bool}
  {Own.size(V, catalog) == Invariant.records(V, catalog) : Nat}

def cached_size_exact(V, catalog, valid):
  match catalog:
    case Own.DNil{}: {==}
    case Own.DCon{key, value, +rest, +count}:
      NatFacts.eq_of_is_eq(count, (1n + Invariant.records(V, rest) : Nat),
        BoolFacts.eq_true_of_and_left(
          Nat.is_eq(count, (1n + Invariant.records(V, rest) : Nat)),
          Bool.and(Bool.not(Own.contains(V, rest, key)), Invariant.valid(V, rest)),
          valid))
