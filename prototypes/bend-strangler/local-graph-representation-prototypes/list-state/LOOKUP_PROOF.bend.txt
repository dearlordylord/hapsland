def picked_lookup(
  -V: Data,
  found: Bool,
  value: V,
  fallback: V,
  rest: Own.Dictionary<V>,
  key: Nat
) -> V:
  Own.choose(V, found, value, next => Own.lookup(V, fallback, rest, key))

# Exact approved lookup_after_insert_exact statement, isolated checker entry.
law lookup_after_insert_exact:
  for -V: Data
  for +catalog: Own.Dictionary<V>
  for +key: Nat
  for +value: V
  for fallback: V
  {Own.lookup(V, fallback, Own.set(V, catalog, key, value), key) == value : V}

def Support.lookup_insert_step(V, found, name, key, previous, rest, value, fallback, same, suffix):
  match found:
    case True{}:
      %Equal.sym(Bool, Nat.is_eq(name, key), True{}, same):
        {picked_lookup(V, _, value, fallback, rest, key) == value : V}
      {==}
    case False{}:
      %Equal.sym(Bool, Nat.is_eq(name, key), False{}, same):
        {picked_lookup(V, _, previous, fallback, Own.set(V, rest, key, value), key) == value : V}
      suffix

def lookup_after_insert_exact(V, catalog, key, value, fallback):
  match catalog:
    case Own.DNil{}:
      %NatFacts.is_eq_refl_sym(key):
        {Own.choose(V, _, value, next => fallback) == value : V}
      {==}
    case Own.DCon{+name, previous, +rest, count}:
      +saved_fallback = fallback
      same = BoolProof.bool_reflexive(Nat.is_eq(name, key))
      suffix = lookup_after_insert_exact(V, rest, key, value, saved_fallback)
      Support.lookup_insert_step(V, Nat.is_eq(name, key), name, key, previous, rest, value,
        saved_fallback, same, suffix)
