# Actual BaseMap first-difference proof through arbitrary-width Word operations,
# the 33-bit character route, and structural String induction.
# This entry proves only diff_first_split and its approved character support law.
def empty_left_split(+head: Char, +tail: String) -> {
  Bool.and(
    Patricia.same_prefix(Map.diff(SNil{}, SCon{head, tail}), SNil{}, SCon{head, tail}),
    Bool.not(Patricia.bool_equal(
      Patricia.route(SNil{}, Map.diff(SNil{}, SCon{head, tail})),
      Patricia.route(SCon{head, tail}, Map.diff(SNil{}, SCon{head, tail}))
    ))
  ) == True{} : Bool
}:
  match head:
    case Chr{value}: {==}

def empty_right_split(+head: Char, +tail: String) -> {

# Actual comparator/bitwise bridge: xor is zero exactly when the words agree.
# Width is structural; this applies to all 32-bit payloads without enumeration.
def comparison_head(+left: Bool, +right: Bool, +tail: Cmp) -> {
  Cmp.is_eq(Word.cmp.fin(left, right, tail))
  == Bool.and(Cmp.is_eq(tail), Cmp.is_eq(Bool.cmp(left, right))) : Bool

  match tail:
    case LT{}: {==}
    case GT{}: {==}
    case EQ{}: {==}

def xor_head(+left: Bool, +right: Bool) -> {

  == Cmp.is_eq(Bool.cmp(left, right)) : Bool

    case False{} False{}: {==}
    case False{} True{}: {==}
    case True{} False{}: {==}
    case True{} True{}: {==}

def prefix_bit_bound(+height: Nat, +position: Nat,
  before: {Nat.is_lt(position, Nat.sub(33n, height)) == True{} : Bool}) -> {
  Nat.is_le(height, Nat.sub(32n, position)) == True{} : Bool
}:
  +sum_lt = Equal.trans(
    Bool,
    Nat.is_lt(Nat.add(height, position), 33n),
    Nat.is_lt(position, Nat.sub(33n, height)),
    True{},
    Equal.sym(
      Bool,
      Nat.is_lt(position, Nat.sub(33n, height)),
      Nat.is_lt(Nat.add(height, position), 33n),
      NatFacts.lt_sub_iff_add_lt(position, height, 33n)
    ),
    before
  )
  +sum_le = NatFacts.le_of_lt_succ(Nat.add(height, position), 32n, sum_lt)
  projection_33 = {n => Nat.is_le(n, 32n) : Nat -> Bool}
  +swapped_le = Equal.trans(
    Bool,
    Nat.is_le(Nat.add(position, height), 32n),
    Nat.is_le(Nat.add(height, position), 32n),
    True{},
    Equal.cong(
      Nat,
      Bool,
      projection_33,
      Nat.add(position, height),
      Nat.add(height, position),
      NatFacts.add_comm(position, height)
    ),
    sum_le
  )
  BoolFacts.eq_true_of_and_right(
    Nat.is_le(position, 32n),
    Nat.is_le(height, Nat.sub(32n, position)),
    Equal.trans(
      Bool,
      Bool.and(Nat.is_le(position, 32n), Nat.is_le(height, Nat.sub(32n, position))),
      Nat.is_le(Nat.add(position, height), 32n),
      True{},
      NatFacts.le_and_le_sub_iff_add_le(position, height, 32n),
      swapped_le
    )
  )
