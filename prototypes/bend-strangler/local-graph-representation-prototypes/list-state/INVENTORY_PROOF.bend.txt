def Inventory.weight_bounded(seen, amount):
  match seen:
    case True{}: NatFacts.zero_le(amount)
    case False{}: NatFacts.le_refl(amount)

def Inventory.unseen_declaration_bounded(visited, declaration):
  match declaration:
    case None{}: {==}
    case Some{Core.Declaration{_, identity, _, _, references}}:
      Inventory.weight_bounded(
        Core.has_identity(visited, identity),
        List.length(&2, Core.Reference, references)
      )

def Inventory.unseen_catalog_bounded(visited, catalog):
  match catalog:
    case Own.DNil{}: {==}
    case Own.DCon{_, +declaration, +rest, _}:
      head = Inventory.unseen_declaration_bounded(visited, declaration)
      suffix = Inventory.unseen_catalog_bounded(visited, rest)
      NatFacts.add_le_add(
        Rank.unseen_declaration(visited, declaration),
        Rank.unseen_catalog(visited, rest),
        Core.declaration_reference_count(declaration),
        Core.trie_reference_count(rest),
        head,
        suffix
      )
