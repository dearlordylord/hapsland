# Internal proof model of the leaf selected by actual BaseMap.seek.
# No canonical insertion or membership theorem is implemented by this definition.
def selected_choice(
  side: Bool,
  left: Maybe<&2, String>,
  right: Maybe<&2, String>
) -> Maybe<&2, String>:
  match side:
    case False{}: left
    case True{}: right

def selected(-V: Data, tree: Map<&2, V>, +key: String) -> Maybe<&2, String>:
  match tree:
    case MTip{}: None{}
    case MLeaf{name, value}: Some{name}
    case MNode{position, lo, hi}:
      selected_choice(Patricia.route(key, position), selected(V, lo, key), selected(V, hi, key))

def seek_empty(-V: Data, +key: String) -> {
  Map.seek(&2, V, MTip{}, key)
  == Tuple{MTip{}, Tuple{key, None{}}}
  : Map<&2, V> & String & Maybe<&2, String>
}:
  {==}

def seek_leaf(-V: Data, +name: String, +value: V, +key: String) -> {
  Map.seek(&2, V, MLeaf{name, value}, key)
  == Tuple{MLeaf{name, value}, Tuple{key, Some{name}}}
  : Map<&2, V> & String & Maybe<&2, String>
}:
  {==}

def seek_node_step(
  -V: Data,
  +position: Nat,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  +key: String,
  side: Bool,
  left: {
    Map.seek(&2, V, lo, key)
    == Tuple{lo, Tuple{key, selected(V, lo, key)}}
    : Map<&2, V> & String & Maybe<&2, String>
  },
  right: {
    Map.seek(&2, V, hi, key)
    == Tuple{hi, Tuple{key, selected(V, hi, key)}}
    : Map<&2, V> & String & Maybe<&2, String>
  }
) -> {
  Map.seek.go(&2, V, MNode{position, lo, hi}, Tuple{key, side})
  == Tuple{MNode{position, lo, hi}, Tuple{key, selected_choice(side,
    selected(V, lo, key), selected(V, hi, key))}} : Map<&2, V> & String & Maybe<&2, String>
}:
  match lo hi side:
    case MTip{} hi False{}: {==}
    case MLeaf{name, value} hi False{}: {==}
    case MNode{q, c, d} hi False{}:
      %Equal.sym(Map<&2, V> & String & Maybe<&2, String>,
        Map.seek(&2, V, MNode{q, c, d}, key),
        Tuple{MNode{q, c, d}, Tuple{key, selected(V, MNode{q, c, d}, key)}}, left):
        {
          Map.lo(&2, V, String & Maybe<&2, String>, position, hi, _)
          == Tuple{
            MNode{position, MNode{q, c, d}, hi},
            Tuple{key, selected(V, MNode{q, c, d}, key)}
          }
          : Map<&2, V> & String & Maybe<&2, String>
        }
      {==}
    case lo MTip{} True{}: {==}
    case lo MLeaf{name, value} True{}: {==}
    case lo MNode{q, c, d} True{}:
      %Equal.sym(Map<&2, V> & String & Maybe<&2, String>,
        Map.seek(&2, V, MNode{q, c, d}, key),
        Tuple{MNode{q, c, d}, Tuple{key, selected(V, MNode{q, c, d}, key)}}, right):
        {
          Map.hi(&2, V, String & Maybe<&2, String>, position, lo, _)
          == Tuple{
            MNode{position, lo, MNode{q, c, d}},
            Tuple{key, selected(V, MNode{q, c, d}, key)}
          }
          : Map<&2, V> & String & Maybe<&2, String>
        }
      {==}

def seek_exact(-V: Data, tree: Map<&2, V>, +key: String) -> {
  Map.seek(&2, V, tree, key)
  == Tuple{tree, Tuple{key, selected(V, tree, key)}}
  : Map<&2, V> & String & Maybe<&2, String>
}:
  match tree:
    case MTip{}: {==}
    case MLeaf{name, value}: {==}
    case MNode{+position, +lo, +hi}:
      %Equal.sym(String & Bool, Map.bit(key, position), Tuple{key, Patricia.route(key, position)},
        BitProof.bit_preserves_key(key, position)):
        {Map.seek.go(&2, V, MNode{position, lo, hi}, _)
          == Tuple{MNode{position, lo, hi}, Tuple{key, selected(V, MNode{position, lo, hi}, key)}}
          : Map<&2, V> & String & Maybe<&2, String>}
      seek_node_step(
        V,
        position,
        lo,
        hi,
        key,
        Patricia.route(key, position),
        seek_exact(V, lo, key),
        seek_exact(V, hi, key)
      )

def node_left_nonempty(-V: Data, +position: Nat, +lo: Map<&2, V>, +hi: Map<&2, V>,
  valid: {Patricia.canonical(V, MNode{position, lo, hi}) == True{} : Bool}) -> {
  Patricia.nonempty(V, lo) == True{} : Bool
}:
  BoolFacts.eq_true_of_and_left(
    Patricia.nonempty(V, lo),
    Bool.and(
      Patricia.nonempty(V, hi),
      Bool.and(
        Patricia.root_after(V, position, lo),
        Bool.and(
          Patricia.root_after(V, position, hi),
          Bool.and(
            Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position),
            Bool.and(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              )
            )
          )
        )
      )
    ),
    valid
  )

def selected_choice_present(side: Bool, +left: Maybe<&2, String>, +right: Maybe<&2, String>,
  left_present: {Patricia.first_leaf_some(left) == True{} : Bool},
  right_present: {Patricia.first_leaf_some(right) == True{} : Bool}) -> {
  Patricia.first_leaf_some(selected_choice(side, left, right)) == True{} : Bool
}:
  match side:
    case False{}: left_present
    case True{}: right_present

def selected_present(-V: Data, tree: Map<&2, V>, +key: String,
  valid: {Patricia.canonical(V, tree) == True{} : Bool}) -> {
  Patricia.first_leaf_some(selected(V, tree, key)) == Patricia.nonempty(V, tree) : Bool
}:
  match tree:
    case MTip{}: {==}
    case MLeaf{name, value}: {==}
    case MNode{+position, +lo, +hi}:
      +valid = valid
      selected_choice_present(
        Patricia.route(key, position),
        selected(V, lo, key),
        selected(V, hi, key),
        Equal.trans(
          Bool,
          Patricia.first_leaf_some(selected(V, lo, key)),
          Patricia.nonempty(V, lo),
          True{},
          selected_present(V, lo, key, node_left_canonical(V, position, lo, hi, valid)),
          node_left_nonempty(V, position, lo, hi, valid)
        ),
        Equal.trans(
          Bool,
          Patricia.first_leaf_some(selected(V, hi, key)),
          Patricia.nonempty(V, hi),
          True{},
          selected_present(V, hi, key, node_right_canonical(V, position, lo, hi, valid)),
          node_right_nonempty(V, position, lo, hi, valid)
        )
      )

def selected_prefix(found: Maybe<&2, String>, +pivot: String, +position: Nat) -> Bool:
  match found:
    case None{}: True{}
    case Some{name}: Patricia.same_prefix(position, pivot, name)

def selected_prefix_choice(
  branch: Bool,
  +left: Maybe<&2, String>,
  +right: Maybe<&2, String>,
  +pivot: String,
  +position: Nat,
  l: {selected_prefix(left, pivot, position) == True{} : Bool},
  r: {selected_prefix(right, pivot, position) == True{} : Bool}

  selected_prefix(selected_choice(branch, left, right), pivot, position) == True{} : Bool
}:
  match branch:
    case False{}: l

def selected_prefix_all(-V: Data, tree: Map<&2, V>, +key: String, +pivot: String, +position: Nat,
  all: {Patricia.all_prefix(V, tree, pivot, position) == True{} : Bool}) -> {
  selected_prefix(selected(V, tree, key), pivot, position) == True{} : Bool
}:
  match tree:
    case MTip{}: {==}
    case MLeaf{name, value}: all
    case MNode{branch, +lo, +hi}:
      +all = all
      selected_prefix_choice(
        Patricia.route(key, branch),
        selected(V, lo, key),
        selected(V, hi, key),
        pivot,
        position,
        selected_prefix_all(
          V,
          lo,
          key,
          pivot,
          position,
          BoolFacts.eq_true_of_and_left(
            Patricia.all_prefix(V, lo, pivot, position),
            Patricia.all_prefix(V, hi, pivot, position),
            all
          )
        ),
        selected_prefix_all(
          V,
          hi,
          key,
          pivot,
          position,
          BoolFacts.eq_true_of_and_right(
            Patricia.all_prefix(V, lo, pivot, position),
            Patricia.all_prefix(V, hi, pivot, position),
            all
          )
        )
      )

def bit_equality_trans(left: Bool, middle: Bool, right: Bool,
  lm: {Patricia.bool_equal(left, middle) == True{} : Bool},
  mr: {Patricia.bool_equal(middle, right) == True{} : Bool}) -> {
  Patricia.bool_equal(left, right) == True{} : Bool
}:
  match left middle right:
    case False{} False{} right: mr
    case True{} True{} right: mr
    case False{} True{} False{}: {==}
    case True{} False{} True{}: {==}
    case False{} True{} True{}: lm
    case True{} False{} False{}: lm

def prefix_trans(count: Nat, +left: String, +middle: String, +right: String,
  lm: {Patricia.same_prefix(count, left, middle) == True{} : Bool},
  mr: {Patricia.same_prefix(count, middle, right) == True{} : Bool}) -> {
  Patricia.same_prefix(count, left, right) == True{} : Bool
}:
  match count:
    case 0n: {==}
    case 1n++rest:
      +lm = lm
      +mr = mr
      BoolFacts.and_eq_true(
        Patricia.same_prefix(rest, left, right),
        Patricia.bool_equal(Patricia.route(left, rest), Patricia.route(right, rest)),
        prefix_trans(
          rest,
          left,
          middle,
          right,
          BoolFacts.eq_true_of_and_left(
            Patricia.same_prefix(rest, left, middle),
            Patricia.bool_equal(Patricia.route(left, rest), Patricia.route(middle, rest)),
            lm
          ),
          BoolFacts.eq_true_of_and_left(
            Patricia.same_prefix(rest, middle, right),
            Patricia.bool_equal(Patricia.route(middle, rest), Patricia.route(right, rest)),
            mr
          )
        ),
        bit_equality_trans(
          Patricia.route(left, rest),
          Patricia.route(middle, rest),
          Patricia.route(right, rest),
          BoolFacts.eq_true_of_and_right(
            Patricia.same_prefix(rest, left, middle),
            Patricia.bool_equal(Patricia.route(left, rest), Patricia.route(middle, rest)),
            lm
          ),
          BoolFacts.eq_true_of_and_right(
            Patricia.same_prefix(rest, middle, right),
            Patricia.bool_equal(Patricia.route(middle, rest), Patricia.route(right, rest)),
            mr
          )
        )
      )

def all_prefix_rebase(-V: Data, tree: Map<&2, V>, +old: String, +fresh: String, +position: Nat,
  +prefix: {Patricia.same_prefix(position, fresh, old) == True{} : Bool},
  all: {Patricia.all_prefix(V, tree, old, position) == True{} : Bool}) -> {
  Patricia.all_prefix(V, tree, fresh, position) == True{} : Bool
}:
  match tree:
    case MTip{}: {==}
    case MLeaf{name, value}: prefix_trans(position, fresh, old, name, prefix, all)
    case MNode{branch, +lo, +hi}:
      +all = all
      BoolFacts.and_eq_true(
        Patricia.all_prefix(V, lo, fresh, position),
        Patricia.all_prefix(V, hi, fresh, position),
        all_prefix_rebase(
          V,
          lo,
          old,
          fresh,
          position,
          prefix,
          BoolFacts.eq_true_of_and_left(
            Patricia.all_prefix(V, lo, old, position),
            Patricia.all_prefix(V, hi, old, position),
            all
          )
        ),
        all_prefix_rebase(
          V,
          hi,
          old,
          fresh,
          position,
          prefix,
          BoolFacts.eq_true_of_and_right(
            Patricia.all_prefix(V, lo, old, position),
            Patricia.all_prefix(V, hi, old, position),
            all
          )
        )
      )

def comparison_order(result: (String & String) & Cmp) -> Cmp:
  match result:
    case Tuple{_, order}: order

def comparison_head_step(
  order: Cmp,
  +left_head: Char,
  +right_head: Char,
  +left: String,
  +right: String,
  suffix: {
    String.cmp(left, right)
    == Tuple{Tuple{left, right}, comparison_order(String.cmp(left, right))}
    : (String & String) & Cmp
  }
) -> {
    String.cmp.fin(left, right, Tuple{Tuple{left_head, right_head}, order})
    == Tuple{
      Tuple{SCon{left_head, left}, SCon{right_head, right}},
      comparison_order(String.cmp.fin(left, right, Tuple{Tuple{left_head, right_head}, order}))
    }
    : (String & String) & Cmp
  }:
  match order:
    case LT{}: {==}
    case GT{}: {==}
    case EQ{}:
      %Equal.sym((String & String) & Cmp, String.cmp(left, right),
        Tuple{Tuple{left, right}, comparison_order(String.cmp(left, right))}, suffix):
        {String.cmp.rec(left_head, right_head, _)
          == Tuple{Tuple{SCon{left_head, left}, SCon{right_head, right}},
            comparison_order(String.cmp.rec(left_head, right_head, _))} : (String & String) & Cmp}
      {==}

def comparison_preserves_keys(left: String, right: String) -> {
  String.cmp(left, right)
  == Tuple{Tuple{left, right}, comparison_order(String.cmp(left, right))}
  : (String & String) & Cmp
}:
  match left right:
    case SNil{} SNil{}: {==}
    case SNil{} SCon{head, tail}: {==}
    case SCon{head, tail} SNil{}: {==}
    case SCon{Chr{+x}, +xs} SCon{Chr{+y}, +ys}:
      comparison_head_step(U32.cmp(x, y), Chr{x}, Chr{y}, xs, ys, comparison_preserves_keys(xs, ys))

    BoolFacts.eq_true_of_and_left(prefix, different, split),
    different_is_opposite(
      Patricia.route(key, Map.diff(key, old)),
      Patricia.route(old, Map.diff(key, old)),
      BoolFacts.eq_true_of_and_right(prefix, different, split)
    ),
    {==}
  )

def prefix_drop(extra: Nat, +lower: Nat, +left: String, +right: String,
  known: {Patricia.same_prefix(Nat.add(lower, extra), left, right) == True{} : Bool}) -> {
  Patricia.same_prefix(lower, left, right) == True{} : Bool
}:
  match extra:
    case 0n:
      %NatFacts.add_zero(lower): {Patricia.same_prefix(_, left, right) == True{} : Bool}
      known
    case 1n++previous:
      +smaller = Nat.add(lower, previous)
      project : Nat -> Bool = n => Patricia.same_prefix(n, left, right)
      +expanded = Equal.trans(
        Bool,
        Patricia.same_prefix((1n + smaller : Nat), left, right),
        Patricia.same_prefix(Nat.add(lower, (1n + previous : Nat)), left, right),
        True{},
        Equal.cong(
          Nat,
          Bool,
          project,
          (1n + smaller : Nat),
          Nat.add(lower, (1n + previous : Nat)),
          Equal.sym(
            Nat,
            Nat.add(lower, (1n + previous : Nat)),
            (1n + smaller : Nat),
            NatFacts.add_succ(lower, previous)
          )
        ),
        known
      )
      prefix_drop(
        previous,
        lower,
        left,
        right,
        BoolFacts.eq_true_of_and_left(
          Patricia.same_prefix(smaller, left, right),
          Patricia.bool_equal(Patricia.route(left, smaller), Patricia.route(right, smaller)),
          expanded
        )
      )

def prefix_shorten(+lower: Nat, +upper: Nat, +left: String, +right: String,
  bounded: {Nat.is_le(lower, upper) == True{} : Bool},
  known: {Patricia.same_prefix(upper, left, right) == True{} : Bool}) -> {
  Patricia.same_prefix(lower, left, right) == True{} : Bool
}:
  project : Nat -> Bool = n => Patricia.same_prefix(n, left, right)
  prefix_drop(Nat.sub(upper, lower), lower, left, right,
    Equal.trans(Bool, Patricia.same_prefix(Nat.add(lower, Nat.sub(upper, lower)), left, right),
      Patricia.same_prefix(upper, left, right), True{},
      Equal.cong(Nat, Bool, project, Nat.add(lower, Nat.sub(upper, lower)), upper,
        DiffProof.ordered_difference(lower, upper, bounded)), known))

def prefix_bit_at(+count: Nat, +position: Nat, +left: String, +right: String,
  before: {Nat.is_lt(position, count) == True{} : Bool},
  known: {Patricia.same_prefix(count, left, right) == True{} : Bool}) -> {
    Patricia.bool_equal(Patricia.route(left, position), Patricia.route(right, position))
    == True{}
    : Bool
  }:
  BoolFacts.eq_true_of_and_right(
    Patricia.same_prefix(position, left, right),
    Patricia.bool_equal(Patricia.route(left, position), Patricia.route(right, position)),
    prefix_shorten(
      (1n + position : Nat),
      count,
      left,
      right,
      NatFacts.succ_le_of_lt(position, count, before),
      known
    )
  )

def all_prefix_shorten(-V: Data, tree: Map<&2, V>, +pivot: String, +lower: Nat, +upper: Nat,

            Patricia.all_prefix(V, lo, pivot, upper),

            known
          )
        )
      )

def selected_prefix_rebase(
  -V: Data,
  +tree: Map<&2, V>,
  +key: String,
  +selected_key: String,
  +pivot: String,
  +count: Nat,
  witness: {selected(V, tree, key) == Some{selected_key} : Maybe<&2, String>},
  +all: {Patricia.all_prefix(V, tree, pivot, count) == True{} : Bool}
) -> {
  Patricia.all_prefix(V, tree, selected_key, count) == True{} : Bool
}:
  project : Maybe<&2, String> -> Bool = found => selected_prefix(found, pivot, count)
  +joined = Equal.trans(
    Bool,
    selected_prefix(Some{selected_key}, pivot, count),
    selected_prefix(selected(V, tree, key), pivot, count),
    True{},
    Equal.cong(
      Maybe<&2, String>,
      Bool,
      project,
      Some{selected_key},
      selected(V, tree, key),
      Equal.sym(Maybe<&2, String>, selected(V, tree, key), Some{selected_key}, witness)
    ),
    selected_prefix_all(V, tree, key, pivot, count, all)
  )
  all_prefix_rebase(
    V,
    tree,
    pivot,
    selected_key,
    count,
    Equal.trans(
      Bool,
      Patricia.same_prefix(count, selected_key, pivot),
      Patricia.same_prefix(count, pivot, selected_key),
      True{},
      prefix_symmetric(count, selected_key, pivot),
      joined
    ),
    all
  )

def above_node_splice_canonical(
  -V: Data,
  +position: Nat,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  +key: String,
  +value: V,
  +old: String,
  witness: {selected(V, MNode{position, lo, hi}, key) == Some{old} : Maybe<&2, String>},
  distinct: {String.eq(key, old) == False{} : Bool},
  +before: {Nat.is_lt(Map.diff(key, old), position) == True{} : Bool},
  +valid: {Patricia.canonical(V, MNode{position, lo, hi}) == True{} : Bool}
) -> {
    Patricia.canonical(
      V,
      Map.ins.splice(&2, V, Map.diff(key, old), key, value, MNode{position, lo, hi})
    )
    == True{}
    : Bool
  }:
  +d = Map.diff(key, old)
  +tree = {MNode{position, lo, hi} : Map<&2, V>}
  +common = node_selected_common_prefix(V, position, lo, hi, key, old, witness, valid)
  +split = DiffProof.diff_first_split(key, old, distinct)
  +prefix = Patricia.same_prefix(d, key, old)
  +different = Bool.not(Patricia.bool_equal(Patricia.route(key, d), Patricia.route(old, d)))
  splice_canonical_nonempty(
    V,
    tree,
    key,
    value,
    d,
    {==},
    before,
    all_prefix_rebase(
      V,
      tree,
      old,
      key,
      d,
      BoolFacts.eq_true_of_and_left(prefix, different, split),
      all_prefix_shorten(V, tree, old, d, position, NatFacts.le_of_lt(d, position, before), common)
    ),
    all_side_retarget(
      V,
      tree,
      d,
      Patricia.route(old, d),
      Bool.not(Patricia.route(key, d)),
      different_is_opposite(
        Patricia.route(key, d),
        Patricia.route(old, d),
        BoolFacts.eq_true_of_and_right(prefix, different, split)
      ),
      all_prefix_side(V, tree, old, d, position, before, common)
    ),
    valid
  )

def selected_node_side_branch(
  -V: Data,
  branch: Bool,
  +position: Nat,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  +key: String,
  +old: String,
  test: {Patricia.route(key, position) == branch : Bool},
  witness: {
    selected_choice(branch, selected(V, lo, key), selected(V, hi, key))
    == Some{old}
    : Maybe<&2, String>
  },
  left: {Patricia.all_side(V, lo, position, False{}) == True{} : Bool},
  right: {Patricia.all_side(V, hi, position, True{}) == True{} : Bool}
) -> {
  Patricia.bool_equal(Patricia.route(key, position), Patricia.route(old, position)) == True{} : Bool
}:
  match branch:
    case False{}:
      %Equal.sym(Bool, Patricia.route(key, position), False{}, test): {
        Patricia.bool_equal(_, Patricia.route(old, position))
        == True{}
        : Bool

      key,
      value,
      position,
      ancestor,
      after
    )
    case MTip{} hi False{} True{}: before
    case MLeaf{name, previous} hi False{} True{}: before
    case MNode{q, a, b} hi False{} True{}: before
    case lo MTip{} True{} True{}: before
    case lo MLeaf{name, previous} True{} True{}: before
    case lo MNode{q, a, b} True{} True{}: before

def insert_root_after(-V: Data, tree: Map<&2, V>, +key: String, +value: V, +position: Nat,
  +ancestor: Nat,
  before: {Patricia.root_after(V, ancestor, tree) == True{} : Bool},
  after: {Nat.is_lt(ancestor, position) == True{} : Bool}) -> {
  Patricia.root_after(V, ancestor, Map.ins(&2, V, tree, key, value, position)) == True{} : Bool
}:
  match tree:
    case MTip{}: {==}
    case MLeaf{name, previous}: splice_root_after(
      V,
      MLeaf{name, previous},
      key,
      value,
      position,
      ancestor,
      after
    )
    case MNode{+branch, +lo, +hi}:
      %Equal.sym(
        String & Bool,
        Map.bit(key, branch),
        Tuple{key, Patricia.route(key, branch)},
        BitProof.bit_preserves_key(key, branch)
      ):
        {
          Patricia.root_after(
            V,
            ancestor,
            Map.ins.go(
              &2,
              V,
              MNode{branch, lo, hi},
              value,
              position,
              _,
              Nat.is_lt(branch, position)
            )
          )
          == True{}
          : Bool
        }
      insert_node_root_after(
        V,
        branch,
        lo,
        hi,
        key,
        value,
        position,
        Patricia.route(key, branch),
        Nat.is_lt(branch, position),
        ancestor,
        before,
        after
      )

def canonical_node_rebuild(
  -V: Data,
  +position: Nat,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  left_nonempty: {Patricia.nonempty(V, lo) == True{} : Bool},
  right_nonempty: {Patricia.nonempty(V, hi) == True{} : Bool},
  left_after: {Patricia.root_after(V, position, lo) == True{} : Bool},
  right_after: {Patricia.root_after(V, position, hi) == True{} : Bool},
  prefix: {
    Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position)
    == True{}
    : Bool
  },
  left_side: {Patricia.all_side(V, lo, position, False{}) == True{} : Bool},
  right_side: {Patricia.all_side(V, hi, position, True{}) == True{} : Bool},
  left_valid: {Patricia.canonical(V, lo) == True{} : Bool},
  right_valid: {Patricia.canonical(V, hi) == True{} : Bool}
) -> {
  Patricia.canonical(V, MNode{position, lo, hi}) == True{} : Bool
}:
  BoolFacts.and_eq_true(
    Patricia.nonempty(V, lo),
    Bool.and(
      Patricia.nonempty(V, hi),
      Bool.and(
        Patricia.root_after(V, position, lo),
        Bool.and(
          Patricia.root_after(V, position, hi),
          Bool.and(
            Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position),
            Bool.and(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              )
            )
          )
        )
      )
    ),
    left_nonempty,
    BoolFacts.and_eq_true(
      Patricia.nonempty(V, hi),
      Bool.and(
        Patricia.root_after(V, position, lo),
        Bool.and(
          Patricia.root_after(V, position, hi),
          Bool.and(
            Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position),
            Bool.and(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              )
            )
          )
        )
      ),
      right_nonempty,
      BoolFacts.and_eq_true(
        Patricia.root_after(V, position, lo),
        Bool.and(
          Patricia.root_after(V, position, hi),
          Bool.and(
            Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position),
            Bool.and(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              )
            )
          )
        ),
        left_after,
        BoolFacts.and_eq_true(
          Patricia.root_after(V, position, hi),
          Bool.and(
            Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position),
            Bool.and(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              )
            )
          ),
          right_after,
          BoolFacts.and_eq_true(
            Patricia.prefix_compatible(V, Patricia.first_leaf(V, lo), lo, hi, position),
            Bool.and(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              )
            ),
            prefix,
            BoolFacts.and_eq_true(
              Patricia.all_side(V, lo, position, False{}),
              Bool.and(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi))
              ),
              left_side,
              BoolFacts.and_eq_true(
                Patricia.all_side(V, hi, position, True{}),
                Bool.and(Patricia.canonical(V, lo), Patricia.canonical(V, hi)),
                right_side,
                BoolFacts.and_eq_true(
                  Patricia.canonical(V, lo),
                  Patricia.canonical(V, hi),
                  left_valid,
                  right_valid
                )
              )
            )
          )
        )
      )
    )
  )

def compatible_from_common_found(
  -V: Data,
  found: Maybe<&2, String>,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  +pivot: String,
  +position: Nat,
  present: {Patricia.first_leaf_some(found) == True{} : Bool},
  selected_prefix_proof: {selected_prefix(found, pivot, position) == True{} : Bool},
  left: {Patricia.all_prefix(V, lo, pivot, position) == True{} : Bool},
  right: {Patricia.all_prefix(V, hi, pivot, position) == True{} : Bool}
) -> {
  Patricia.prefix_compatible(V, found, lo, hi, position) == True{} : Bool
}:
  match found:
    case None{}:
      Empty.absurd({False{} == True{} : Bool}, false_true_impossible(present))
    case Some{+fresh}:
      +same = Equal.trans(
        Bool,
        Patricia.same_prefix(position, fresh, pivot),
        Patricia.same_prefix(position, pivot, fresh),
        True{},
        prefix_symmetric(position, fresh, pivot),
        selected_prefix_proof
      )
      BoolFacts.and_eq_true(
        Patricia.all_prefix(V, lo, fresh, position),
        Patricia.all_prefix(V, hi, fresh, position),
        all_prefix_rebase(V, lo, pivot, fresh, position, same, left),
        all_prefix_rebase(V, hi, pivot, fresh, position, same, right)
      )

    right

def canonical_insert_left(-V: Data, +position: Nat, +lo: Map<&2, V>, +hi: Map<&2, V>,
  +key: String, +value: V, +split: Nat, +pivot: String,
  +valid: {Patricia.canonical(V, MNode{position, lo, hi}) == True{} : Bool},
  +changed: {Patricia.canonical(V, Map.ins(&2, V, lo, key, value, split)) == True{} : Bool},
  +after: {Nat.is_lt(position, split) == True{} : Bool},
  key_prefix: {Patricia.same_prefix(position, pivot, key) == True{} : Bool},
  key_side: {Patricia.bool_equal(Patricia.route(key, position), False{}) == True{} : Bool},
  +left_prefix: {Patricia.all_prefix(V, lo, pivot, position) == True{} : Bool},
  +right_prefix: {Patricia.all_prefix(V, hi, pivot, position) == True{} : Bool}) -> {
  Patricia.canonical(V, MNode{position, Map.ins(&2, V, lo, key, value, split), hi}) == True{} : Bool
}:
  +next_prefix = insert_preserves_prefix(
    V,
    lo,
    key,
    value,
    split,
    pivot,
    position,
    key_prefix,
    left_prefix
  )
  +next_nonempty = insert_nonempty(V, lo, key, value, split)
  +next_after = insert_root_after(
    V,
    lo,
    key,
    value,
    split,
    position,
    node_left_after(V, position, lo, hi, valid),
    after
  )
  +next_side = insert_preserves_side(
    V,
    lo,
    key,
    value,
    split,
    position,
    False{},
    key_side,
    node_left_side(V, position, lo, hi, valid)
  )
  canonical_node_rebuild(
    V,
    position,
    Map.ins(&2, V, lo, key, value, split),
    hi,
    next_nonempty,
    node_right_nonempty(V, position, lo, hi, valid),
    next_after,
    node_right_after(V, position, lo, hi, valid),
    compatible_from_common(
      V,
      Map.ins(&2, V, lo, key, value, split),
      hi,
      pivot,
      position,
      changed,
      next_nonempty,
      next_prefix,
      right_prefix
    ),
    next_side,
    node_right_side(V, position, lo, hi, valid),
    changed,
    node_right_canonical(V, position, lo, hi, valid)
  )

def inserted_prefix_below_split(+position: Nat, +key: String, +old: String,
  distinct: {String.eq(key, old) == False{} : Bool},

    Bool,
    Patricia.same_prefix(position, old, key),

    prefix_shorten(position, d, key, old, NatFacts.le_of_lt(position, d, before), prefix)
  )

def selected_name(value: Maybe<&2, String>) -> String:

    case Some{name}: name

def selected_none_impossible(

    Equal.cong(Maybe<&2, String>, Bool, Patricia.first_leaf_some, None{}, Some{old}, witness)

def insertion_canonical_node(

  +value: V,

  side: Bool,
  deeper: Bool,

  +depth_test: {Nat.is_lt(position, Map.diff(key, old)) == deeper : Bool},
  +witness: {

    == Some{old}

  +distinct: {String.eq(key, old) == False{} : Bool},
  +valid: {Patricia.canonical(V, MNode{position, lo, hi}) == True{} : Bool},

  right: {
    selected(V, hi, key)
    == Some{old}
    : Maybe<&2, String>

      V,

        &2,
        V,

        value,
        Map.diff(key, old),

    == True{}
    : Bool

      canonical_above_actual(

        position,
        lo,

            Patricia.all_prefix(V, hi, old, position),

          BoolFacts.eq_true_of_and_right(
            Patricia.all_prefix(V, lo, old, position),
            Patricia.all_prefix(V, hi, old, position),
            common
          )
        )
      )
    case True{} True{}:
      +actual = selected_node_from_test(V, position, lo, hi, key, old, True{}, side_test, witness)
      +common = node_selected_common_prefix(V, position, lo, hi, key, old, actual, valid)
      canonical_deep_right_actual(
        V,
        position,
        lo,
        hi,
        key,
        value,

        True{},
        {==},
        {==},

          lo,

          right(witness),
          depth_test,
          inserted_prefix_below_split(position, key, old, distinct, depth_test),
          true_side_exact(Patricia.route(key, position), side_test),
          BoolFacts.eq_true_of_and_left(
            Patricia.all_prefix(V, lo, old, position),
            Patricia.all_prefix(V, hi, old, position),
            common
          ),
          BoolFacts.eq_true_of_and_right(

def insertion_canonical_selected(-V: Data, tree: Map<&2, V>, +key: String, +value: V, +old: String,
  witness: {selected(V, tree, key) == Some{old} : Maybe<&2, String>},
  +distinct: {String.eq(key, old) == False{} : Bool},
  +valid: {Patricia.canonical(V, tree) == True{} : Bool}) -> {
  Patricia.canonical(V, Map.ins(&2, V, tree, key, value, Map.diff(key, old))) == True{} : Bool
}:
  match tree:
    case MTip{}:
      Empty.absurd(
        {
          Patricia.canonical(V, Map.ins(&2, V, MTip{}, key, value, Map.diff(key, old)))
          == True{}
          : Bool
        },
        selected_none_impossible(old, witness)
      )
    case MLeaf{+name, +previous}:
      same = Equal.cong(Maybe<&2, String>, String, selected_name, Some{name}, Some{old}, witness)
      %Equal.sym(String, name, old, same): {
        Patricia.canonical(V, Map.ins(&2, V, MLeaf{_, previous}, key, value, Map.diff(key, old)))
        == True{}
        : Bool
      }
      leaf_splice_canonical(V, old, previous, key, value, distinct)
    case MNode{+position, +lo, +hi}:
      left : {selected(V, lo, key) == Some{old} : Maybe<&2, String>} -> {
        Patricia.canonical(V, Map.ins(&2, V, lo, key, value, Map.diff(key, old)))
        == True{}
        : Bool
      } = child =>
        insertion_canonical_selected(
          V,
          lo,
          key,
          value,
          old,
          child,
          distinct,
          node_left_canonical(V, position, lo, hi, valid)
        )
      right : {selected(V, hi, key) == Some{old} : Maybe<&2, String>} -> {
        Patricia.canonical(V, Map.ins(&2, V, hi, key, value, Map.diff(key, old)))
        == True{}
        : Bool
      } = child =>
        insertion_canonical_selected(
          V,
          hi,
          key,
          value,
          old,
          child,
          distinct,
          node_right_canonical(V, position, lo, hi, valid)
        )
      %Equal.sym(
        String & Bool,
        Map.bit(key, position),
        Tuple{key, Patricia.route(key, position)},
        BitProof.bit_preserves_key(key, position)
      ):
        {
          Patricia.canonical(
            V,
            Map.ins.go(
              &2,
              V,
              MNode{position, lo, hi},
              value,
              Map.diff(key, old),
              _,
              Nat.is_lt(position, Map.diff(key, old))
            )
          )
          == True{}
          : Bool
        }
      insertion_canonical_node(
        V,
        position,
        lo,
        hi,
        key,
        value,
        old,
        Patricia.route(key, position),
        Nat.is_lt(position, Map.diff(key, old)),
        {==},
        {==},
        witness,
        distinct,
        valid,
        left,
        right
      )

def put_keys_node(
  ~A: Data,
  ~B: Data,
  ~test: A -> B -> String -> Bool,
  -V: Data,
  +position: Nat,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  +key: String,
  +value: V,
  side: Bool,
  +a: A,
  +b: B,
  left_present: {Patricia.nonempty(V, lo) == True{} : Bool},
  right_present: {Patricia.nonempty(V, hi) == True{} : Bool},
  left: {all_keys(~A, ~B, ~test, V, lo, a, b) == True{} : Bool},
  right: {all_keys(~A, ~B, ~test, V, hi, a, b) == True{} : Bool},
  next_left: {all_keys(~A, ~B, ~test, V, Map.put(&2, V, lo, key, value), a, b) == True{} : Bool},
  next_right: {all_keys(~A, ~B, ~test, V, Map.put(&2, V, hi, key, value), a, b) == True{} : Bool}
) -> {
    all_keys(
      ~A,
      ~B,
      ~test,
      V,
      Map.put.go(&2, V, MNode{position, lo, hi}, value, Tuple{key, side}),
      a,
      b
    )
    == True{}
    : Bool
  }:
  match lo hi side:
    case MTip{} hi False{}:
      Empty.absurd(
        {
          all_keys(
            ~A,
            ~B,
            ~test,
            V,
            Map.put.go(&2, V, MNode{position, MTip{}, hi}, value, Tuple{key, False{}}),
            a,
            b
          )
          == True{}
          : Bool
        },
        false_true_impossible(left_present)
      )
    case MLeaf{name, previous} hi False{}:
      BoolFacts.and_eq_true(test(a, b, name), all_keys(~A, ~B, ~test, V, hi, a, b), left, right)
    case MNode{branch, c, d} hi False{}:
      BoolFacts.and_eq_true(
        all_keys(~A, ~B, ~test, V, Map.put(&2, V, MNode{branch, c, d}, key, value), a, b),
        all_keys(~A, ~B, ~test, V, hi, a, b),
        next_left,
        right
      )
    case lo MTip{} True{}:
      Empty.absurd(
        {
          all_keys(
            ~A,
            ~B,
            ~test,
            V,
            Map.put.go(&2, V, MNode{position, lo, MTip{}}, value, Tuple{key, True{}}),
            a,
            b
          )
          == True{}
          : Bool
        },
        false_true_impossible(right_present)
      )
    case lo MLeaf{name, previous} True{}:
      BoolFacts.and_eq_true(all_keys(~A, ~B, ~test, V, lo, a, b), test(a, b, name), left, right)
    case lo MNode{branch, c, d} True{}:
      BoolFacts.and_eq_true(
        all_keys(~A, ~B, ~test, V, lo, a, b),
        all_keys(~A, ~B, ~test, V, Map.put(&2, V, MNode{branch, c, d}, key, value), a, b),
        left,
        next_right
      )

def put_keys(~A: Data, ~B: Data, ~test: A -> B -> String -> Bool, -V: Data,
  tree: Map<&2, V>, +key: String, +value: V, +a: A, +b: B,
  +valid: {Patricia.canonical(V, tree) == True{} : Bool},
  present: {Patricia.nonempty(V, tree) == True{} : Bool},
  +old: {all_keys(~A, ~B, ~test, V, tree, a, b) == True{} : Bool}) -> {
  all_keys(~A, ~B, ~test, V, Map.put(&2, V, tree, key, value), a, b) == True{} : Bool
}:
  match tree:
    case MTip{}:
      Empty.absurd({test(a, b, key) == True{} : Bool}, false_true_impossible(present))
    case MLeaf{name, previous}: old
    case MNode{+position, +lo, +hi}:
      +left = BoolFacts.eq_true_of_and_left(
        all_keys(~A, ~B, ~test, V, lo, a, b),
        all_keys(~A, ~B, ~test, V, hi, a, b),
        old
      )
      +right = BoolFacts.eq_true_of_and_right(
        all_keys(~A, ~B, ~test, V, lo, a, b),
        all_keys(~A, ~B, ~test, V, hi, a, b),
        old
      )
      +lp = node_left_nonempty(V, position, lo, hi, valid)
      +rp = node_right_nonempty(V, position, lo, hi, valid)
      %Equal.sym(
        String & Bool,
        Map.bit(key, position),
        Tuple{key, Patricia.route(key, position)},
        BitProof.bit_preserves_key(key, position)
      ):
        {
          all_keys(~A, ~B, ~test, V, Map.put.go(&2, V, MNode{position, lo, hi}, value, _), a, b)
          == True{}
          : Bool
        }
      put_keys_node(
        ~A,
        ~B,
        ~test,
        V,
        position,
        lo,
        hi,
        key,
        value,
        Patricia.route(key, position),
        a,
        b,
        lp,
        rp,
        left,
        right,
        put_keys(
          ~A,
          ~B,
          ~test,
          V,
          lo,
          key,
          value,
          a,
          b,
          node_left_canonical(V, position, lo, hi, valid),
          lp,
          left
        ),
        put_keys(
          ~A,
          ~B,
          ~test,
          V,
          hi,
          key,
          value,
          a,
          b,
          node_right_canonical(V, position, lo, hi, valid),
          rp,
          right
        )
      )

    node_right_side(V, position, lo, hi, valid)
  )
  canonical_node_rebuild(
    V,
    position,
    lo,
    Map.put(&2, V, hi, key, value),
    node_left_nonempty(V, position, lo, hi, valid),
    next_nonempty,
    node_left_after(V, position, lo, hi, valid),
    next_after,
    compatible_from_common(
      V,
      lo,
      Map.put(&2, V, hi, key, value),
      pivot,
      position,
      node_left_canonical(V, position, lo, hi, valid),
      node_left_nonempty(V, position, lo, hi, valid),
      left_prefix,
      next_prefix
    ),
    node_left_side(V, position, lo, hi, valid),
    next_side,
    node_left_canonical(V, position, lo, hi, valid),
    changed
  )

    Bool,

def has_node_exact(

          == Tuple{
            MNode{position, MNode{branch, a, b}, hi},
            selected_equal(selected(V, MNode{branch, a, b}, key), key)
          }
          : Map<&2, V> & Bool
        }
      {==}
    case lo MTip{} True{}: {==}
    case lo MLeaf{name, value} True{}:
      %Equal.sym(
        Map<&2, V> & Bool,
        Map.has(&2, V, MLeaf{name, value}, key),

            MNode{position, lo, MLeaf{name, value}},

          Map.hi(&2, V, Bool, position, lo, _)
          == Tuple{
            MNode{position, lo, MNode{branch, a, b}},
            selected_equal(selected(V, MNode{branch, a, b}, key), key)
          }
          : Map<&2, V> & Bool
        }
      {==}

def boolean_equal_eq(
  +left: Bool,
  +right: Bool,
  same: {Patricia.bool_equal(left, right) == True{} : Bool}
) -> {left == right : Bool}:
  match left right:
    case False{} False{}: {==}
    case True{} True{}: {==}
    case False{} True{}: Empty.absurd({False{} == True{} : Bool}, false_true_impossible(same))
    case True{} False{}: Empty.absurd({True{} == False{} : Bool}, false_true_impossible(same))

def opposite_impossible(side: Bool, same: {Bool.not(side) == side : Bool}) -> Empty:
  match side:
    case False{}: false_true_impossible(Equal.sym(Bool, True{}, False{}, same))
    case True{}: false_true_impossible(same)

def opposite_key_excluded(+query: String, +name: String, +position: Nat, +side: Bool,
  query_side: {Patricia.route(query, position) == Bool.not(side) : Bool},
  name_side: {Patricia.bool_equal(Patricia.route(name, position), side) == True{} : Bool}) -> {
  String.eq(query, name) == False{} : Bool
}:
  refute : {String.eq(query, name) == True{} : Bool} -> Empty = same =>
    route_at : String -> Bool = text => Patricia.route(text, position)
    opposite_impossible(
      side,
      Equal.trans(
        Bool,
        Bool.not(side),
        Patricia.route(query, position),
        side,
        Equal.sym(Bool, Patricia.route(query, position), Bool.not(side), query_side),
        Equal.trans(
          Bool,
          Patricia.route(query, position),
          Patricia.route(name, position),
          side,
          Equal.cong(
            String,
            Bool,
            route_at,
            query,
            name,
            StringFacts.eq_of_eq_true(query, name, same)
          ),
          boolean_equal_eq(Patricia.route(name, position), side, name_side)
        )
      )
    )
  false_of_refuted(String.eq(query, name), refute)

def all_side_excludes(-V: Data, tree: Map<&2, V>, +query: String, +position: Nat, +side: Bool,
  +query_side: {Patricia.route(query, position) == Bool.not(side) : Bool},

        leaf_exists(V, lo, query),

          lo,

            Patricia.all_side(V, lo, position, side),

            sided

        hi,
        query,
        position,
        side,
        query_side,
        BoolFacts.eq_true_of_and_right(
          Patricia.all_side(V, lo, position, side),

          sided
        )
      )

def canonical_selected_node_membership(
  -V: Data,
  +position: Nat,
  +lo: Map<&2, V>,
  +hi: Map<&2, V>,
  +query: String,
  side: Bool,
  test: {Patricia.route(query, position) == side : Bool},
  valid: {Patricia.canonical(V, MNode{position, lo, hi}) == True{} : Bool},
  left: {selected_equal(selected(V, lo, query), query) == leaf_exists(V, lo, query) : Bool},
  right: {selected_equal(selected(V, hi, query), query) == leaf_exists(V, hi, query) : Bool}
) -> {
  selected_equal(selected_choice(side, selected(V, lo, query), selected(V, hi, query)), query)
    == Bool.or(leaf_exists(V, lo, query), leaf_exists(V, hi, query)) : Bool

  match side:
    case False{}:
      %Equal.sym(
        Bool,
        leaf_exists(V, hi, query),
        False{},
        all_side_excludes(
          V,
          hi,
          query,
          position,
          True{},
          test,
          node_right_side(V, position, lo, hi, valid)
        )
      ):
        {
          selected_equal(selected(V, lo, query), query)
          == Bool.or(leaf_exists(V, lo, query), _)
          : Bool
        }
      Equal.trans(
        Bool,
        selected_equal(selected(V, lo, query), query),
        leaf_exists(V, lo, query),
        Bool.or(leaf_exists(V, lo, query), False{}),
        left,
        Equal.sym(
          Bool,
          Bool.or(leaf_exists(V, lo, query), False{}),
          leaf_exists(V, lo, query),
          BoolFacts.or_false(leaf_exists(V, lo, query))
        )
      )
    case True{}:
      %Equal.sym(
        Bool,
        leaf_exists(V, lo, query),
        False{},
        all_side_excludes(
          V,
          lo,
          query,
          position,
          False{},
          test,
          node_left_side(V, position, lo, hi, valid)
        )
      ):
        {
          selected_equal(selected(V, hi, query), query)
          == Bool.or(_, leaf_exists(V, hi, query))
          : Bool
        }
      right

def canonical_selected_membership(-V: Data, tree: Map<&2, V>, +query: String,
  +valid: {Patricia.canonical(V, tree) == True{} : Bool}) -> {
  selected_equal(selected(V, tree, query), query) == leaf_exists(V, tree, query) : Bool

    case lo MTip{} True{} True{}:
      %Equal.sym(

        leaf_exists(V, Map.ins(&2, V, MTip{}, key, value, position), query),
        Bool.or(String.eq(query, key), leaf_exists(V, MTip{}, query)),
        right

          Bool.or(leaf_exists(V, lo, query), _)
          == Bool.or(
            String.eq(query, key),
            Bool.or(leaf_exists(V, lo, query), leaf_exists(V, MTip{}, query))

          : Bool
        }
      or_middle_front(
        leaf_exists(V, lo, query),
        String.eq(query, key),
        leaf_exists(V, MTip{}, query)
      )
    case lo MLeaf{name, previous} True{} True{}:
      %Equal.sym(
        Bool,
        leaf_exists(V, Map.ins(&2, V, MLeaf{name, previous}, key, value, position), query),
