def choice_result(-V: Data, found: Bool, previous: V, fallback: V) -> V:
  Own.choose(V, found, previous, next => fallback)

law other_lookup_insert_step:
  for -V: Data
  for +found: Bool
  for +name: Nat
  for +key: Nat
  for +other: Nat
  for +previous: V
  for +rest: Own.Dictionary<V>
  for +value: V
  for +fallback: V
  for +same: {Nat.is_eq(name, key) == found : Bool}
  for +distinct: {Nat.is_eq(key, other) == False{} : Bool}
  for suffix: {
    Own.lookup(V, fallback, Own.set(V, rest, key, value), other)
    == Own.lookup(V, fallback, rest, other)
    : V
  }
  {Own.lookup(V, fallback, Helpers.insert_head(V, found, name, previous, rest, key, value), other)
    == Own.lookup(V, fallback, Own.cons(V, name, previous, rest), other) : V}

# Constructor membership has ordinary Boolean disjunction semantics.

law other_lookup_preserved:
  for -V: Data
  for +catalog: Own.Dictionary<V>
  for +key: Nat
  for +other: Nat
  for value: V
  for +fallback: V
  for distinct: {Nat.is_eq(key, other) == False{} : Bool}
  {Own.lookup(V, fallback, Own.set(V, catalog, key, value), other)
    == Own.lookup(V, fallback, catalog, other) : V}

# Exactly the prior memberships plus the inserted key; no extra identities.

def other_lookup_insert_step(
  V,
  found,
  name,
  key,
  other,
  previous,
  rest,
  value,
  fallback,
  same,
  distinct,
  suffix
):
  match found:
    case True{}:
      +head_missing = Helpers.head_distinct(name, key, other, same, distinct)
      %Equal.sym(Bool, Nat.is_eq(name, other), False{}, head_missing):
        {
          choice_result(V, _, value, Own.lookup(V, fallback, rest, other))
          == Own.lookup(V, fallback, Own.cons(V, name, previous, rest), other)
          : V
        }
      %Equal.sym(Bool, Nat.is_eq(name, other), False{}, head_missing):
        {
          Own.lookup(V, fallback, rest, other)
          == choice_result(V, _, previous, Own.lookup(V, fallback, rest, other))
          : V
        }
      {==}
    case False{}:
      %suffix:
        {
          choice_result(
            V,
            Nat.is_eq(name, other),
            previous,
            Own.lookup(V, fallback, Own.set(V, rest, key, value), other)
          )
          == choice_result(V, Nat.is_eq(name, other), previous, _)
          : V
        }
      {==}

def other_lookup_preserved(V, catalog, key, other, value, fallback, distinct):
  match catalog:
    case Own.DNil{}:
      %Equal.sym(Bool, Nat.is_eq(key, other), False{}, distinct):
        {choice_result(V, _, value, fallback) == fallback : V}
      {==}
    case Own.DCon{+name, previous, +rest, count}:
      +saved_value = value
      +saved_distinct = distinct
      same = Helpers.bool_reflexive(Nat.is_eq(name, key))
      suffix = other_lookup_preserved(V, rest, key, other, saved_value, fallback, saved_distinct)
      other_lookup_insert_step(
        V,
        Nat.is_eq(name, key),
        name,
        key,
        other,
        previous,
        rest,
        saved_value,
        fallback,
        same,
        saved_distinct,
        suffix
      )
