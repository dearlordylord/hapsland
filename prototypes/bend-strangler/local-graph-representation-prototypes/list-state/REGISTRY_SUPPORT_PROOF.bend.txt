import Base
import ./REGISTRY_RELATION.bend as Registry
import ./REGISTRY_SUPPORT_LAWS.bend as Support
import ../../../../vendor/bendlib/packages/bend-mathlib/bool.bend as BoolFacts
import ../../../../vendor/bendlib/packages/bend-mathlib/string.bend as StringFacts

def present(-V: Data, found: Maybe<&2, V>) -> Bool:
  match found:
    case None{}: False{}
    case Some{_}: True{}

def extract(-V: Data, fallback: V, found: Maybe<&2, V>) -> V:
  match found:
    case None{}: fallback
    case Some{value}: value

def get_type(names: List<&2, String>, index: Nat, name: String) -> Data:
  {Registry.get(names, index) == Some{name} : Maybe<&2, String>}

def decode_type(names: List<&2, String>, identity: Nat, name: String) -> Data:
  {Registry.decode(names, identity) == Some{name} : Maybe<&2, String>}

def some_string(name: String) -> Maybe<&2, String>:
  Some{name}

def Support.none_not_some(V, value, equal):
  BoolFacts.internal_false_ne_true(
    Equal.cong(Maybe<&2, V>, Bool, present(V), None{}, Some{value}, equal)
  )

def Support.some_injective(V, left, right, equal):
  Equal.cong(Maybe<&2, V>, V, extract(V, left), Some{left}, Some{right}, equal)

def Support.find_increment_get_step(found, head, rest, name, index, find_matches, selected, suffix):
  match found:
    case None{}:
      Empty.absurd(get_type(head <> rest, index, name), Support.none_not_some(Nat, index, selected))
    case Some{+position}:
      %Support.some_injective(Nat, (1n + position : Nat), index, selected):
        {Registry.get(head <> rest, _) == Some{name} : Maybe<&2, String>}
      suffix(position, find_matches)

def Support.find_get_step(equal, head, rest, name, index, matches, selected, suffix):
  match equal:
    case True{}:
      %Support.some_injective(Nat, 0n, index, selected):
        {Registry.get(head <> rest, _) == Some{name} : Maybe<&2, String>}
      Equal.cong(
        String,
        Maybe<&2, String>,
        some_string,
        head,
        name,
        StringFacts.eq_of_eq_true(head, name, matches)
      )
    case False{}:
      Support.find_increment_get_step(
        Registry.find(rest, name),
        head,
        rest,
        name,
        index,
        {==},
        selected,
        suffix
      )

def Support.get_from_find(names, name, index, selected):
  match names:
    case []:
      Empty.absurd(get_type([], index, name), Support.none_not_some(Nat, index, selected))
    case +head <> +rest:
      continue = {+position => next_selected => Support.get_from_find(rest, name, position, next_selected) : Support.get_continuation(rest, name)}
      Support.find_get_step(
        String.eq(head, name),
        head,
        rest,
        name,
        index,
        {==},
        selected,
        continue
      )

def Support.increment_decodes(found, names, name, identity, selected, suffix):
  match found:
    case None{}:
      Empty.absurd(
        decode_type(names, identity, name),
        Support.none_not_some(Nat, identity, selected)
      )
    case Some{+index}:
      %Support.some_injective(Nat, (1n + index : Nat), identity, selected):
        {Registry.decode(names, _) == Some{name} : Maybe<&2, String>}
      suffix(index, {==})
