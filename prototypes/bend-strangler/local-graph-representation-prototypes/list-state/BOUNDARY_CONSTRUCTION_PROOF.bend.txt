def Support.all_present_collect(items, names):
  match items:
    case []: {==}
    case +head <> +rest:
      +after = Registry.names_of(Registry.intern(names, head))
      +collected = Registry.collect(rest, after)
      BoolFacts.and_eq_true(
        Registry.present(collected, head),
        Encoding.all_present(rest, collected),
        Completion.collect_head_present(rest, names, head),
        Support.all_present_collect(rest, after)
      )

def Support.encoded_if_exact(valid, names, input):
  match valid input:
    case False{} _: {==}
    case True{} Encoding.Input{_, _, _, _, _, _}: {==}

def Support.covered_id_step(found, names, name, matches, covered):
  match found:
    case None{}:
      Empty.absurd(
        {Registry.decode(names, 0n) == Some{name} : Maybe<&2, String>},
        BoolFacts.internal_false_ne_true(covered)
      )
    case Some{identity}: RegistryLaws.selected_id_decodes(names, name, identity, matches)

def Support.covered_range_step(found, names, name, matches, covered):
  match found:
    case None{}:
      Empty.absurd(
        {
          Bool.and(Nat.is_lt(0n, 0n), Nat.is_le(0n, List.length(&2, String, names)))
          == True{}
          : Bool
        },
        BoolFacts.internal_false_ne_true(covered)
      )
    case Some{identity}: RegistryLaws.assigned_id_in_range(names, name, identity, matches)

def covered_identity_exact(names, name, covered):
  Support.covered_id_step(Registry.encode(names, name), names, name, {==}, covered)

def covered_identity_in_range(names, name, covered):
  Support.covered_range_step(Registry.encode(names, name), names, name, {==}, covered)

def Support.covered_text_exact(names, name, covered):
  Equal.cong(
    Maybe<&2, String>,
    String,
    Encoding.decoded_string,
    Registry.decode(names, Encoding.identity(names, name)),
    Some{name},
    covered_identity_exact(names, name, covered)
  )

def boundary_encoding_complete(input):
  +source = input
  +items = Encoding.input_strings(source)
  +names = Registry.registry(items)
  +unique = Registry.unique(names)
  +covered = Encoding.all_present(items, names)
  +valid = Bool.and(unique, covered)
  exact = BoolFacts.and_eq_true(unique, covered,
    RegistryLaws.constructed_registry_unique(items), Support.all_present_collect(items, []))
  Equal.trans(Bool, Encoding.is_encoded(Encoding.encode_if(valid, names, source)), valid, True{},
    Support.encoded_if_exact(valid, names, source), exact)

def lookup_key_roundtrip(names, name, aware, expected, covered):
  match aware expected:
    case False{} _:
      Support.covered_text_exact(
        names,
        name,
        BoolFacts.eq_true_of_and_left(
          Registry.present(names, name),
          Encoding.all_present(
            [String.append("type:", name), String.append("function:", name)],
            names
          ),
          covered
        )
      )
    case True{} Source.TypeKind{}:
      Support.covered_text_exact(
        names,
        String.append("type:", name),
        BoolFacts.eq_true_of_and_left(
          Registry.present(names, String.append("type:", name)),
          Encoding.all_present([String.append("function:", name)], names),
          BoolFacts.eq_true_of_and_right(
            Registry.present(names, name),
            Encoding.all_present(
              [String.append("type:", name), String.append("function:", name)],
              names
            ),
            covered
          )
        )
      )
    case True{} Source.AnyKind{}:
      Support.covered_text_exact(
        names,
        String.append("function:", name),
        BoolFacts.eq_true_of_and_left(
          Registry.present(names, String.append("function:", name)),
          True{},
          BoolFacts.eq_true_of_and_right(
            Registry.present(names, String.append("type:", name)),
            Encoding.all_present([String.append("function:", name)], names),
            BoolFacts.eq_true_of_and_right(
              Registry.present(names, name),
              Encoding.all_present(
                [String.append("type:", name), String.append("function:", name)],
                names
              ),
              covered
            )
          )
        )
      )
    case True{} Source.FunctionKind{}:
      Support.covered_text_exact(
        names,
        String.append("function:", name),
        BoolFacts.eq_true_of_and_left(
          Registry.present(names, String.append("function:", name)),
          True{},
          BoolFacts.eq_true_of_and_right(
            Registry.present(names, String.append("type:", name)),
            Encoding.all_present([String.append("function:", name)], names),
            BoolFacts.eq_true_of_and_right(
              Registry.present(names, name),
              Encoding.all_present(
                [String.append("type:", name), String.append("function:", name)],
                names
              ),
              covered
            )
          )
        )
      )

def import_composite_exact(names, path, binding, covered):
  match binding:
    case Source.ImportBinding{+import_path, +name, +type_only}:
      +composite = SourceTraversal.import_target_key(
        path,
        Source.ImportBinding{import_path, name, type_only}
      )
      covered_identity_exact(
        names,
        composite,
        BoolFacts.eq_true_of_and_left(
          Registry.present(names, composite),
          True{},
          BoolFacts.eq_true_of_and_right(
            Registry.present(names, name),
            Encoding.all_present([composite], names),
            BoolFacts.eq_true_of_and_right(
              Registry.present(names, import_path),
              Encoding.all_present([name, composite], names),
              covered
            )
          )
        )
      )
