def equal_type(left: Maybe<&2, Nat>, right: Maybe<&2, Nat>) -> Data:
  {left == right : Maybe<&2, Nat>}

def selected_type(found: Maybe<&2, Nat>, equal: Bool, length: Nat, identity: Nat) -> Data:
  {
    Registry.increment(Append.append_lookup(found, equal, length))
    == Some{identity}
    : Maybe<&2, Nat>
  }

def some_nat(index: Nat) -> Maybe<&2, Nat>:
  Some{index}

def Append.increment_injective(left, right, equal):
  match left right:
    case None{} None{}: {==}
    case None{} Some{index}:
      Empty.absurd(
        equal_type(None{}, Some{index}),
        Support.none_not_some(Nat, (1n + index : Nat), equal)
      )
    case Some{index} None{}:
      Empty.absurd(equal_type(Some{index}, None{}), Support.none_not_some(Nat, (1n + index : Nat),
        Equal.sym(Maybe<&2, Nat>, Some{(1n + index : Nat)}, None{}, equal)))
    case Some{+a} Some{+b}:
      Equal.cong(Nat, Maybe<&2, Nat>, some_nat, a, b,
        NatFacts.succ_inj(a, b, Support.some_injective(Nat, (1n + a : Nat), (1n + b : Nat), equal)))

def Append.appended_get_exact(names, name):
  match names:
    case []: {==}
    case head <> rest: Append.appended_get_exact(rest, name)

def Append.append_increment(found, equal, length):
  match found equal:
    case None{} True{}: {==}
    case None{} False{}: {==}
    case Some{index} _: {==}

def Append.append_empty(equal):
  match equal:
    case True{}: {==}
    case False{}: {==}

def Append.append_head_step(equal, rest, added, query, suffix):
  match equal:
    case True{}: {==}
    case False{}:
      +before = Registry.find(rest, query)
      +after = Registry.find(List.append(&2, String, rest, [added]), query)
      +matches = String.eq(added, query)
      +length = List.length(&2, String, rest)
      +specified = Append.append_lookup(before, matches, length)
      Equal.trans(
        Maybe<&2, Nat>,
        Registry.increment(after),
        Registry.increment(specified),
        Append.append_lookup(Registry.increment(before), matches, (1n + length : Nat)),
        Equal.cong(Maybe<&2, Nat>, Maybe<&2, Nat>, Registry.increment, after, specified, suffix),
        Equal.sym(
          Maybe<&2, Nat>,
          Append.append_lookup(Registry.increment(before), matches, (1n + length : Nat)),
          Registry.increment(specified),
          Append.append_increment(before, matches, length)
        )
      )

def Append.find_append_exact(names, added, query):
  match names:
    case []: Append.append_empty(String.eq(added, query))
    case +head <> +rest:
      Append.append_head_step(
        String.eq(head, query),
        rest,
        added,
        query,
        Append.find_append_exact(rest, added, query)
      )

def Append.append_encoded_selected(found, equal, length, identity, selected):
  match found:
    case None{}:
      Empty.absurd(
        selected_type(None{}, equal, length, identity),
        Support.none_not_some(Nat, identity, selected)
      )
    case Some{index}: selected

def append_for_query(+name: String, length: Nat, found: Maybe<&2, Nat>) -> Maybe<&2, Nat>:
  Append.append_lookup(found, String.eq(name, name), length)

def append_missing_at(length: Nat, equal: Bool) -> Maybe<&2, Nat>:
  Append.append_missing(equal, length)

def Append.appended_encode_missing(names, name, missing):
  +found = Registry.find(names, name)
  +length = List.length(&2, String, names)
  found_missing = Append.increment_injective(found, None{}, missing)
  +after = Registry.find(List.append(&2, String, names, [name]), name)
  +specified = Append.append_lookup(found, String.eq(name, name), length)
  +absent = Append.append_lookup(None{}, String.eq(name, name), length)
  first = Append.find_append_exact(names, name, name)
  second = Equal.cong(
    Maybe<&2, Nat>,
    Maybe<&2, Nat>,
    append_for_query(name, length),
    found,
    None{},
    found_missing
  )
  third = Equal.cong(
    Bool,
    Maybe<&2, Nat>,
    append_missing_at(length),
    String.eq(name, name),
    True{},
    StringFacts.eq_refl(name)
  )
  combined = Equal.trans(Maybe<&2, Nat>, after, specified, Some{length}, first,
    Equal.trans(Maybe<&2, Nat>, specified, absent, Some{length}, second, third))
  Equal.cong(Maybe<&2, Nat>, Maybe<&2, Nat>, Registry.increment, after, Some{length}, combined)
