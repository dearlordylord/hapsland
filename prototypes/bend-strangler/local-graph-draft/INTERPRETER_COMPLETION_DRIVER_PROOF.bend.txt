# Whole sibling sequencing uses actual child-returned state. Its monotonicity
# comes from the proved full interpreter, including failed partial execution.
def references_child(fuel: Nat, +facts: Core.Facts, +path: String,
    +owner: Nat, +index: Nat, +depth: Nat, +tail: List<&2, Core.Reference>,
    state: Spec.State) -> Spec.ReferencesResult:
  Spec.as_references(Spec.interpret(fuel,
    Spec.ReferencesRequest{tail, facts, path, owner, index, depth, state}))

def references_dispatch_complete(+fuel: Nat, +facts: Core.Facts, +path: String,
    +owner: Nat, +index: Nat, +depth: Nat, +items: List<&2, Core.Reference>,
    +state: Spec.State, +result: Spec.DispatchResult,
    complete: {Measure.projected_success(fuel,
      Spec.ReferencesRequest{items, facts, path, owner, index, depth, state}, result)
      == True{} : Bool}) -> {
  C.references_complete(fuel, owner, index, Spec.as_references(result)) == True{} : Bool
}:
  match result:
    case Spec.ListResult{Spec.ReferencesResult{state, references}}: complete
    case Spec.ListResult{Spec.ReferencesFailure{state}}: complete
    case Spec.BuildResult{Spec.Missing{state}}: complete
    case Spec.BuildResult{Spec.Built{state, tree}}: complete
    case Spec.BuildResult{Spec.InsufficientBound{state}}: complete

def prepend_complete(
  +fuel: Nat,
  +owner: Nat,
  +index: Nat,
  +head: Spec.TreeReference,
  +tail: Spec.ReferencesResult,
  head_complete: {
    Measure.flat_success(Spec.flat_head(head, owner, index, Spec.flatten(fuel)))
    == True{}
    : Bool
  },
  tail_complete: {C.references_complete(fuel, owner, (index + 1n : Nat), tail) == True{} : Bool}
) -> {
  C.references_complete((1n + fuel : Nat), owner, index, Spec.prepend(tail, head)) == True{} : Bool
}:
  match tail:
    case Spec.ReferencesFailure{state}:
      Empty.absurd({False{} == True{} : Bool}, Order.internal_false_ne_true(tail_complete))
    case Spec.ReferencesResult{state, references}:
      Flat.references_complete(fuel, owner, index, head, references, head_complete, tail_complete)

def after_reference_complete(
  +fuel: Nat,
  +facts: Core.Facts,
  +path: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +tail: List<&2, Core.Reference>,
  +before: Spec.State,
  +result: Spec.ReferenceResult,
  head_complete: {C.reference_complete(fuel, owner, index, result) == True{} : Bool},
  progress: {
    R.holds(
      Prim.reference_visited(result),
      Measure.visited(before),
      Rank.declaration_catalog(facts)
    )
    == True{}
    : Bool
  },
  enough: {
    Nat.is_le(
      (3n * Rank.unseen_catalog(
        Measure.visited(before),
        Rank.declaration_catalog(facts)
      ) + 3n * (1n + List.length(&2, Core.Reference, tail) : Nat) + 1n : Nat),
      (1n + fuel : Nat)
    )
    == True{}
    : Bool
  },
  sibling: C.contract(fuel)
) -> {
  C.references_complete((1n + fuel : Nat), owner, index,
    Spec.after_reference(result, references_child(fuel, facts, path, owner,
      (index + 1n : Nat), depth, tail))) == True{} : Bool
}:
  match result:
    case Spec.ReferenceFailure{state}:
      Empty.absurd({False{} == True{} : Bool}, Order.internal_false_ne_true(head_complete))
    case Spec.ReferenceResult{+state, reference}:
      +progress = progress
      +catalog = Rank.declaration_catalog(facts)
      +after = Rank.unseen_catalog(Measure.visited(state), catalog)
      +prior = Rank.unseen_catalog(Measure.visited(before), catalog)
      +request = {Spec.ReferencesRequest{tail, facts, path, owner, (index + 1n : Nat), depth, state}
        : Spec.Request}
      +returned = Spec.interpret(fuel, request)
      valid = R.lawful(Measure.visited(state), Measure.visited(before), catalog, progress)
      drop = Order.eq_true_of_and_right(Patricia.canonical(Bool, Measure.visited(state)),
        Nat.is_le(after, prior), progress)
      complete = sibling(request, valid,
        RequestRank.reference_sibling_bound(after, prior, List.length(&2, Core.Reference, tail),
          fuel, drop, enough))
      prepend_complete(fuel, owner, index, reference, Spec.as_references(returned), head_complete,
        references_dispatch_complete(fuel, facts, path, owner, (index + 1n : Nat), depth,
          tail, state, returned, complete))

def inventory_child(
  +fuel: Nat,
  +catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +request: Spec.Request,
  valid: {
    Patricia.canonical(Bool, Measure.visited(Measure.request_state(request)))
    == True{}
    : Bool
  }
) -> {
  R.holds(Measure.visited(Measure.result_state(Spec.interpret(fuel, request))),
    Measure.visited(Measure.request_state(request)), catalog) == True{} : Bool
}:
  Inventory.interpret_preserves_inventory(fuel, request, catalog, valid)

def references_complete(+fuel: Nat, +items: List<&2, Core.Reference>,
    +facts: Core.Facts, +path: String, +owner: Nat, +index: Nat, +depth: Nat,
    +state: Spec.State,
    valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool},
    enough: {Nat.is_le(Measure.request_rank(
      Spec.ReferencesRequest{items, facts, path, owner, index, depth, state}),
      (1n + fuel : Nat)) == True{} : Bool},
    child: C.contract(fuel), sibling: C.contract(fuel)) -> {
  C.references_complete((1n + fuel : Nat), owner, index,
    Spec.references(items, facts, path, owner, index, depth, state,
      Spec.interpret(fuel), Spec.interpret(fuel))) == True{} : Bool
}:
  match items:
    case []: {==}
    case +head <> +tail:
      +valid = valid
      +enough = enough
      +catalog = Rank.declaration_catalog(facts)
      +result = Spec.reference(head, facts, path, owner, index, depth, state, C.build_child(fuel))
      head_complete = Reference.reference_complete(
        RequestRank.selected_unseen_rank_bound,
        fuel,
        head,
        facts,
        path,
        owner,
        index,
        depth,
        state,
        valid,
        RequestRank.reference_head_allowance(
          Rank.unseen_catalog(Measure.visited(state), catalog),
          List.length(&2, Core.Reference, tail),
          fuel,
          enough
        ),
        child
      )
      progress = InventoryTransport.reference_progress(fuel, head, facts, path, owner, index,
        depth, state, Measure.visited(state), catalog,
        R.reflexive(Measure.visited(state), catalog, valid), inventory_child(fuel, catalog))
      after_reference_complete(fuel, facts, path, owner, index, depth, tail, state, result,
        head_complete, progress, enough, sibling)

def consume_rank(+rank: Nat, +parent: Nat, +fuel: Nat,
    drop: {Nat.is_le((rank + 1n : Nat), parent) == True{} : Bool},
    enough: {Nat.is_le(parent, (1n + fuel : Nat)) == True{} : Bool}) -> {
  Nat.is_le(rank, fuel) == True{} : Bool
}:
  successor = {%NatFacts.add_comm_sym(1n, rank):
    {Nat.is_le(_, (1n + fuel : Nat)) == True{} : Bool}
    Order.le_trans((rank + 1n : Nat), parent, (1n + fuel : Nat), drop, enough)
    : {Nat.is_le((1n + rank : Nat), (1n + fuel : Nat)) == True{} : Bool}}
  NatFacts.le_of_succ_le_succ(rank, fuel, successor)

def tree_result_complete(+fuel: Nat, +owner: Nat, +artifact: Nat,
    +result: Spec.ReferencesResult,
    complete: {C.references_complete(fuel, owner, 0n, result) == True{} : Bool}) -> {
  C.build_complete((1n + fuel : Nat), Spec.tree_result(result, owner, artifact)) == True{} : Bool
}:
  match result:
    case Spec.ReferencesFailure{state}:
      Empty.absurd({False{} == True{} : Bool}, Order.internal_false_ne_true(complete))
    case Spec.ReferencesResult{state, references}:
      Flat.tree_complete(fuel, owner, artifact, references, complete)

def entered_valid(+state: Spec.State, +path: String, +identity: String,
    valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool}) -> {
  Patricia.canonical(Bool, Measure.visited(Spec.enter(state, path, identity))) == True{} : Bool
}:
  R.lawful(Measure.visited(Spec.enter(state, path, identity)), Measure.visited(state), MTip{},
    Enter.enter_progress(state, path, identity, Measure.visited(state), MTip{},
      R.reflexive(Measure.visited(state), MTip{}, valid)))

def walk_complete(+fuel: Nat, +found: Maybe<&2, Core.Declaration>,
    +facts: Core.Facts, +path: String, +name: String, +expected: Core.ExpectedKind,
    +state: Spec.State, +depth: Nat,
    selected: {Spec.find_declaration(facts, name, expected) == found : Maybe<&2, Core.Declaration>},
    valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool},
    enough: {Nat.is_le(Measure.root_rank(facts, name, expected, Measure.visited(state)),
      (1n + fuel : Nat)) == True{} : Bool},
    child: C.contract(fuel)) -> {
  C.build_complete((1n + fuel : Nat),
    Spec.walk_declaration(found, facts, path, state, depth, Spec.interpret(fuel))) == True{} : Bool
}:
  match found:
    case None{}: {==}
    case Some{Core.Declaration{+artifact, +identity, +function, +bundled, +children}}:
      +declaration = {Core.Declaration{artifact, identity, function, bundled, children} : Core.Declaration}
      +entered = Spec.enter(state, path, identity)
      +owner = Spec.address(state)
      +request = {Spec.ReferencesRequest{children, facts, path, owner, 0n, depth, entered} : Spec.Request}
      +returned = Spec.interpret(fuel, request)
      bound = consume_rank(Measure.request_rank(request),
        Measure.root_rank(facts, name, expected, Measure.visited(state)), fuel,
        RequestRank.selected_children_rank_decreases(facts, path, state, name, expected,
          declaration, depth, selected), enough)
      complete = child(request, entered_valid(state, path, identity, valid), bound)
      tree_result_complete(fuel, owner, artifact, Spec.as_references(returned),
        references_dispatch_complete(fuel, facts, path, owner, 0n, depth, children,
          entered, returned, complete))

def build_projection_complete(+fuel: Nat, +facts: Core.Facts, +path: String,
    +name: String, +expected: Core.ExpectedKind, +depth: Nat, +state: Spec.State,
    +result: Spec.Build,
    complete: {C.build_complete(fuel, result) == True{} : Bool}) -> {
  Measure.projected_success(fuel, Spec.Request{facts, path, name, expected, depth, state},
    Spec.BuildResult{result}) == True{} : Bool
}:
  match result:
    case Spec.Missing{state}: complete
    case Spec.Built{state, tree}: complete
    case Spec.InsufficientBound{state}: complete

def references_projection_complete(+fuel: Nat, +items: List<&2, Core.Reference>,
    +facts: Core.Facts, +path: String, +owner: Nat, +index: Nat, +depth: Nat,
    +state: Spec.State, +result: Spec.ReferencesResult,
    complete: {C.references_complete(fuel, owner, index, result) == True{} : Bool}) -> {
      Measure.projected_success(
        fuel,
        Spec.ReferencesRequest{items, facts, path, owner, index, depth, state},
        Spec.ListResult{result}
      )
      == True{}
      : Bool
    }:
  match result:
    case Spec.ReferencesResult{state, references}: complete
    case Spec.ReferencesFailure{state}: complete

def dispatch_complete(
  +fuel: Nat,
  +request: Spec.Request,
  valid: {
    Patricia.canonical(Bool, Measure.visited(Measure.request_state(request)))
    == True{}
    : Bool
  },
  enough: {Nat.is_le(Measure.request_rank(request), (1n + fuel : Nat)) == True{} : Bool},
  child: C.contract(fuel),
  sibling: C.contract(fuel)
) -> {
  Measure.projected_success((1n + fuel : Nat), request,
    Spec.dispatch(request, Spec.interpret(fuel), Spec.interpret(fuel))) == True{} : Bool
}:
  match request:
    case Spec.Request{+facts, +path, +name, +expected, +depth, +state}:
      build_projection_complete(
        (1n + fuel : Nat),
        facts,
        path,
        name,
        expected,
        depth,
        state,
        Spec.walk_declaration(
          Spec.find_declaration(facts, name, expected),
          facts,
          path,
          state,
          depth,
          Spec.interpret(fuel)
        ),
        walk_complete(
          fuel,
          Spec.find_declaration(facts, name, expected),
          facts,
          path,
          name,
          expected,
          state,
          depth,
          {==},
          valid,
          enough,
          child
        )
      )
    case Spec.ReferencesRequest{+items, +facts, +path, +owner, +index, +depth, +state}:
      references_projection_complete(
        (1n + fuel : Nat),
        items,
        facts,
        path,
        owner,
        index,
        depth,
        state,
        Spec.references(
          items,
          facts,
          path,
          owner,
          index,
          depth,
          state,
          Spec.interpret(fuel),
          Spec.interpret(fuel)
        ),
        references_complete(
          fuel,
          items,
          facts,
          path,
          owner,
          index,
          depth,
          state,
          valid,
          enough,
          child,
          sibling
        )
      )
