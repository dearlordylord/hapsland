# Exact tuple projection, independent of the boundary-history qualification closure.
def identity_projection_exact(pair: Map<&2, Bool> & Bool) -> {
  Core.map_result(Bool, pair) == Insert.has_value(Bool, pair) : Bool
}:
  match pair:
    case Tuple{_, value}: {==}

# Internal proof transports for the single approved selected-child inventory law.
def membership_union(+visited: Map<&2, Bool>, +identity: String, +query: String,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool}) -> {
  CoreTraversal.has_identity(Map.set(&2, Bool, visited, identity, True{}), query)
  == Bool.or(String.eq(query, identity), CoreTraversal.has_identity(visited, query)) : Bool
}:
  +updated = Map.set(&2, Bool, visited, identity, True{})
  append : Bool -> Bool = old => Bool.or(String.eq(query, identity), old)
  Equal.trans(
    Bool,
    CoreTraversal.has_identity(updated, query),
    Insert.has_value(Bool, Map.has(&2, Bool, updated, query)),
    Bool.or(String.eq(query, identity), CoreTraversal.has_identity(visited, query)),
    identity_projection_exact(Map.has(&2, Bool, updated, query)),
    Equal.trans(
      Bool,
      Insert.has_value(Bool, Map.has(&2, Bool, updated, query)),
      Bool.or(
        String.eq(query, identity),
        Insert.has_value(Bool, Map.has(&2, Bool, visited, query))
      ),
      Bool.or(String.eq(query, identity), CoreTraversal.has_identity(visited, query)),
      Insert.set_has_membership(Bool, visited, identity, True{}, query, valid),
      Equal.cong(
        Bool,
        Bool,
        append,
        Insert.has_value(Bool, Map.has(&2, Bool, visited, query)),
        CoreTraversal.has_identity(visited, query),
        Equal.sym(
          Bool,
          CoreTraversal.has_identity(visited, query),
          Insert.has_value(Bool, Map.has(&2, Bool, visited, query)),
          identity_projection_exact(Map.has(&2, Bool, visited, query))
        )
      )
    )
  )

def union_weight_nonincreasing(+seen: Bool, +same: Bool, +amount: Nat) -> {
  Nat.is_le(Rank.unseen_weight(Bool.or(same, seen), amount), Rank.unseen_weight(seen, amount))
  == True{} : Bool
}:
  match seen same:
    case True{} True{}: {==}
    case True{} False{}: {==}
    case False{} True{}: NatFacts.zero_le(amount)
    case False{} False{}: NatFacts.le_refl(amount)

def declaration_nonincreasing(+visited: Map<&2, Bool>, +identity: String,
  +declaration: Maybe<&2, Core.Declaration>,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool}) -> {
  Nat.is_le(Rank.unseen_declaration(Map.set(&2, Bool, visited, identity, True{}), declaration),
    Rank.unseen_declaration(visited, declaration)) == True{} : Bool
}:
  match declaration:
    case None{}: {==}
    case Some{Core.Declaration{_, +name, _, _, +references}}:
      %Equal.sym(Bool,
        CoreTraversal.has_identity(Map.set(&2, Bool, visited, identity, True{}), name),
        Bool.or(String.eq(name, identity), CoreTraversal.has_identity(visited, name)),
        membership_union(visited, identity, name, valid)):
        {
          Nat.is_le(
            Rank.unseen_weight(_, List.length(&2, Core.Reference, references)),
            Rank.unseen_weight(
              CoreTraversal.has_identity(visited, name),
              List.length(&2, Core.Reference, references)
            )
          )
          == True{}
          : Bool
        }
      union_weight_nonincreasing(CoreTraversal.has_identity(visited, name), String.eq(name, identity),
        List.length(&2, Core.Reference, references))

def catalog_nonincreasing(catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +visited: Map<&2, Bool>, +identity: String,
  +valid: {Patricia.canonical(Bool, visited) == True{} : Bool}) -> {
  Nat.is_le(Rank.unseen_catalog(Map.set(&2, Bool, visited, identity, True{}), catalog),
    Rank.unseen_catalog(visited, catalog)) == True{} : Bool
}:
  match catalog:
    case MTip{}: {==}
    case MLeaf{_, declaration}: declaration_nonincreasing(visited, identity, declaration, valid)
    case MNode{_, +lo, +hi}:
      NatFacts.add_le_add(
        Rank.unseen_catalog(Map.set(&2, Bool, visited, identity, True{}), lo),
        Rank.unseen_catalog(Map.set(&2, Bool, visited, identity, True{}), hi),
        Rank.unseen_catalog(visited, lo), Rank.unseen_catalog(visited, hi),
        catalog_nonincreasing(lo, visited, identity, valid),
        catalog_nonincreasing(hi, visited, identity, valid))

def inserted_zero(+visited: Map<&2, Bool>, declaration: Core.Declaration,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool}) -> {
    Rank.unseen_declaration(
      Map.set(&2, Bool, visited, CoreTraversal.declaration_identity(declaration), True{}),
      Some{declaration}
    )
    == 0n
    : Nat
  }:
  match declaration:
    case Core.Declaration{_, +identity, _, _, +references}:
      %Equal.sym(Bool,
        CoreTraversal.has_identity(Map.set(&2, Bool, visited, identity, True{}), identity),
        Bool.or(String.eq(identity, identity), CoreTraversal.has_identity(visited, identity)),
        membership_union(visited, identity, identity, valid)):
        {Rank.unseen_weight(_, List.length(&2, Core.Reference, references)) == 0n : Nat}
      %Equal.sym(Bool, String.eq(identity, identity), True{}, StringFacts.eq_refl(identity)):
        {Rank.unseen_weight(Bool.or(_, CoreTraversal.has_identity(visited, identity)),
          List.length(&2, Core.Reference, references)) == 0n : Nat}
      {==}

def absent_full(
  +visited: Map<&2, Bool>,
  declaration: Core.Declaration,
  absent: {CoreTraversal.has_identity(visited, CoreTraversal.declaration_identity(declaration)) == False{} : Bool}
) -> {
  Rank.unseen_declaration(visited, Some{declaration})
  == CoreTraversal.declaration_reference_count(Some{declaration}) : Nat
}:
  match declaration:
    case Core.Declaration{_, +identity, _, _, +references}:
      %Equal.sym(Bool, CoreTraversal.has_identity(visited, identity), False{}, absent):
        {Rank.unseen_weight(_, List.length(&2, Core.Reference, references))
          == List.length(&2, Core.Reference, references) : Nat}
      {==}

def child_claim(+catalog: Map<&2, Maybe<&2, Core.Declaration>>, +visited: Map<&2, Bool>,
  +declaration: Core.Declaration) -> Bool:
  Nat.is_le(
    (Rank.unseen_catalog(
      Map.set(&2, Bool, visited, CoreTraversal.declaration_identity(declaration), True{}),
      catalog
    ) + CoreTraversal.declaration_reference_count(Some{declaration}) : Nat),
    Rank.unseen_catalog(visited, catalog)
  )

def is_some(value: Maybe<&2, Core.Declaration>) -> Bool:
  match value:
    case None{}: False{}
    case Some{_}: True{}

def none_not_selected(declaration: Core.Declaration,
  selected: {None{} == Some{declaration} : Maybe<&2, Core.Declaration>}) -> Empty:
  BoolFacts.internal_false_ne_true(Equal.cong(Maybe<&2, Core.Declaration>, Bool,
    is_some, None{}, Some{declaration}, selected))

def leaf_child_bound(
  +name: String,
  +value: Maybe<&2, Core.Declaration>,
  +visited: Map<&2, Bool>,
  +declaration: Core.Declaration,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool},
  +selected: {value == Some{declaration} : Maybe<&2, Core.Declaration>},
  absent: {CoreTraversal.has_identity(visited, CoreTraversal.declaration_identity(declaration)) == False{} : Bool}
) -> {
  child_claim(MLeaf{name, value}, visited, declaration) == True{} : Bool
}:
  +updated = Map.set(&2, Bool, visited, CoreTraversal.declaration_identity(declaration), True{})
  +amount = CoreTraversal.declaration_reference_count(Some{declaration})
  %Equal.sym(Maybe<&2, Core.Declaration>, value, Some{declaration}, selected):
    {Nat.is_le((Rank.unseen_declaration(updated, _) + amount : Nat),
      Rank.unseen_declaration(visited, value)) == True{} : Bool}
  %Equal.sym(Maybe<&2, Core.Declaration>, value, Some{declaration}, selected):
    {Nat.is_le((Rank.unseen_declaration(updated, Some{declaration}) + amount : Nat),
      Rank.unseen_declaration(visited, _)) == True{} : Bool}
  %Equal.sym(Nat, Rank.unseen_declaration(updated, Some{declaration}), 0n,
    inserted_zero(visited, declaration, valid)):
    {
      Nat.is_le((_ + amount : Nat), Rank.unseen_declaration(visited, Some{declaration}))
      == True{}
      : Bool
    }
  %Equal.sym(Nat, Rank.unseen_declaration(visited, Some{declaration}), amount,
    absent_full(visited, declaration, absent)):
    {Nat.is_le(amount, _) == True{} : Bool}
  NatFacts.le_refl(amount)

def leaf_lookup_child(
  +name: String,
  +value: Maybe<&2, Core.Declaration>,
  comparison: (String & String) & Cmp,
  +visited: Map<&2, Bool>,
  +declaration: Core.Declaration,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool},
  selected: {
    Core.map_result(
      Maybe<&2, Core.Declaration>,
      Map.get.leaf(Maybe<&2, Core.Declaration>, None{}, value, comparison)
    )
    == Some{declaration}
    : Maybe<&2, Core.Declaration>
  },
  absent: {CoreTraversal.has_identity(visited, CoreTraversal.declaration_identity(declaration)) == False{} : Bool}
) -> {
  child_claim(MLeaf{name, value}, visited, declaration) == True{} : Bool
}:
  match comparison:
    case Tuple{Tuple{_, _}, EQ{}}: leaf_child_bound(
      name,
      value,
      visited,
      declaration,
      valid,
      selected,
      absent
    )
    case Tuple{Tuple{_, _}, LT{}}:
      Empty.absurd({child_claim(MLeaf{name, value}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected))
    case Tuple{Tuple{_, _}, GT{}}:
      Empty.absurd({child_claim(MLeaf{name, value}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected))

def child_left(
  +position: Nat,
  +lo: Map<&2, Maybe<&2, Core.Declaration>>,
  +hi: Map<&2, Maybe<&2, Core.Declaration>>,
  +visited: Map<&2, Bool>,
  +declaration: Core.Declaration,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool},
  child: {child_claim(lo, visited, declaration) == True{} : Bool}
) -> {
  child_claim(MNode{position, lo, hi}, visited, declaration) == True{} : Bool
}:
  +updated = Map.set(&2, Bool, visited, CoreTraversal.declaration_identity(declaration), True{})
  +a = Rank.unseen_catalog(updated, lo)
  +b = Rank.unseen_catalog(updated, hi)
  +c = CoreTraversal.declaration_reference_count(Some{declaration})
  %NatFacts.add_assoc_sym(a, b, c):
    {
      Nat.is_le(_, (Rank.unseen_catalog(visited, lo) + Rank.unseen_catalog(visited, hi) : Nat))
      == True{}
      : Bool
    }
  %NatFacts.add_comm(c, b):
    {
      Nat.is_le(
        (a + _ : Nat),
        (Rank.unseen_catalog(visited, lo) + Rank.unseen_catalog(visited, hi) : Nat)
      )
      == True{}
      : Bool
    }
  %NatFacts.add_assoc(a, c, b):
    {
      Nat.is_le(_, (Rank.unseen_catalog(visited, lo) + Rank.unseen_catalog(visited, hi) : Nat))
      == True{}
      : Bool
    }
  NatFacts.add_le_add(
    (a + c : Nat),
    b,
    Rank.unseen_catalog(visited, lo),
    Rank.unseen_catalog(visited, hi),
    child,
    catalog_nonincreasing(hi, visited, CoreTraversal.declaration_identity(declaration), valid)
  )

  result: Map<&2, Maybe<&2, Core.Declaration>> & Maybe<&2, Core.Declaration>) -> {

def left_selected(
  +position: Nat,
  +sibling: Map<&2, Maybe<&2, Core.Declaration>>,
  result: Map<&2, Maybe<&2, Core.Declaration>> & Maybe<&2, Core.Declaration>,
  +declaration: Core.Declaration,
  selected: {
    Core.map_result(
      Maybe<&2, Core.Declaration>,
      Map.lo(
        &2,
        Maybe<&2, Core.Declaration>,
        Maybe<&2, Core.Declaration>,
        position,
        sibling,
        result
      )
    )
    == Some{declaration}
    : Maybe<&2, Core.Declaration>
  }
) -> {
    Core.map_result(Maybe<&2, Core.Declaration>, result)
    == Some{declaration}
    : Maybe<&2, Core.Declaration>
  }:
  Equal.trans(
    Maybe<&2, Core.Declaration>,
    Core.map_result(Maybe<&2, Core.Declaration>, result),
    Core.map_result(
      Maybe<&2, Core.Declaration>,
      Map.lo(
        &2,
        Maybe<&2, Core.Declaration>,
        Maybe<&2, Core.Declaration>,
        position,
        sibling,
        result
      )
    ),
    Some{declaration},
    Equal.sym(
      Maybe<&2, Core.Declaration>,
      Core.map_result(
        Maybe<&2, Core.Declaration>,
        Map.lo(
          &2,
          Maybe<&2, Core.Declaration>,
          Maybe<&2, Core.Declaration>,
          position,
          sibling,
          result
        )
      ),
      Core.map_result(Maybe<&2, Core.Declaration>, result),
      left_projection(position, sibling, result)
    ),
    selected
  )

def routed_child(
  catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  route: String & Bool,
  +visited: Map<&2, Bool>,
  +declaration: Core.Declaration,
  +valid: {Patricia.canonical(Bool, visited) == True{} : Bool},
  selected: {
    Core.map_result(
      Maybe<&2, Core.Declaration>,
      Map.get.go(Maybe<&2, Core.Declaration>, catalog, None{}, route)
    )
    == Some{declaration}
    : Maybe<&2, Core.Declaration>
  },
  +absent: {CoreTraversal.has_identity(visited, CoreTraversal.declaration_identity(declaration)) == False{} : Bool}
) -> {
  child_claim(catalog, visited, declaration) == True{} : Bool
}:
  match catalog route:
    case MTip{} _:
      Empty.absurd(
        {child_claim(MTip{}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected)
      )
    case MLeaf{+name, +value} _:
      Empty.absurd(
        {child_claim(MLeaf{name, value}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected)
      )
    case MNode{+position, MTip{}, +sibling} Tuple{+key, False{}}:
      Empty.absurd(
        {child_claim(MNode{position, MTip{}, sibling}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected)
      )
    case MNode{+position, MLeaf{+name, +value}, +sibling} Tuple{+key, False{}}:
      child_left(
        position,
        MLeaf{name, value},
        sibling,
        visited,
        declaration,
        valid,
        leaf_lookup_child(
          name,
          value,
          String.cmp(key, name),
          visited,
          declaration,
          valid,
          left_selected(
            position,
            sibling,
            Map.get.leaf(Maybe<&2, Core.Declaration>, None{}, value, String.cmp(key, name)),
            declaration,
            selected
          ),
          absent
        )
      )
    case MNode{+position, MNode{+branch, +a, +b}, +sibling} Tuple{+key, False{}}:
      child_left(
        position,
        MNode{branch, a, b},
        sibling,
        visited,
        declaration,
        valid,
        routed_child(
          MNode{branch, a, b},
          Map.bit(key, branch),
          visited,
          declaration,
          valid,
          left_selected(
            position,
            sibling,
            Map.get.go(
              Maybe<&2, Core.Declaration>,
              MNode{branch, a, b},
              None{},
              Map.bit(key, branch)
            ),
            declaration,
            selected
          ),
          absent
        )
      )
    case MNode{+position, +sibling, MTip{}} Tuple{+key, True{}}:
      Empty.absurd(
        {child_claim(MNode{position, sibling, MTip{}}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected)
      )
    case MNode{+position, +sibling, MLeaf{+name, +value}} Tuple{+key, True{}}:
      child_right(
        position,
        sibling,
        MLeaf{name, value},
        visited,
        declaration,
        valid,
        leaf_lookup_child(
          name,
          value,
          String.cmp(key, name),
          visited,
          declaration,
          valid,
          right_selected(
            position,
            sibling,
            Map.get.leaf(Maybe<&2, Core.Declaration>, None{}, value, String.cmp(key, name)),
            declaration,
            selected
          ),
          absent
        )
      )
    case MNode{+position, +sibling, MNode{+branch, +a, +b}} Tuple{+key, True{}}:
      child_right(
        position,
        sibling,
        MNode{branch, a, b},
        visited,
        declaration,
        valid,
        routed_child(
          MNode{branch, a, b},
          Map.bit(key, branch),
          visited,
          declaration,
          valid,
          right_selected(
            position,
            sibling,
            Map.get.go(
              Maybe<&2, Core.Declaration>,
              MNode{branch, a, b},
              None{},
              Map.bit(key, branch)
            ),
            declaration,
            selected
          ),
          absent
        )
      )

def selected_child_inventory_decrease(+catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +visited: Map<&2, Bool>,
  +key: String,
  +declaration: Core.Declaration,
  valid: {Patricia.canonical(Bool, visited) == True{} : Bool},
  selected: {
  Core.map_result(Maybe<&2, Core.Declaration>,
    Map.get(Maybe<&2, Core.Declaration>, None{}, catalog, key))
  == Some{declaration} : Maybe<&2, Core.Declaration>
},
  absent: {
  CoreTraversal.has_identity(visited, CoreTraversal.declaration_identity(declaration)) == False{} : Bool
}) -> {
  Nat.is_le(
    (Rank.unseen_catalog(
      Map.set(&2, Bool, visited, CoreTraversal.declaration_identity(declaration), True{}), catalog
    ) + CoreTraversal.declaration_reference_count(Some{declaration}) : Nat),
    Rank.unseen_catalog(visited, catalog)
  ) == True{} : Bool
}:
  match catalog:
    case MTip{}:
      Empty.absurd({child_claim(MTip{}, visited, declaration) == True{} : Bool},
        none_not_selected(declaration, selected))
    case MLeaf{+name, +value}:
      leaf_lookup_child(
        name,
        value,
        String.cmp(key, name),
        visited,
        declaration,
        valid,
        selected,
        absent
      )
    case MNode{+position, +lo, +hi}:
      routed_child(
        MNode{position, lo, hi},
        Map.bit(key, position),
        visited,
        declaration,
        valid,
        selected,
        absent
      )
