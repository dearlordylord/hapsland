# Proof-only observation of the actual independent single-reference result.
# Its recursive child remains Spec.interpret; this adds no classifier or transition.
def observation(result: Spec.ReferenceResult, machine: Core.Machine,
    +fuel: Nat, owner: Nat, index: Nat) -> Relation.Observation:
  match result machine:
    case Spec.ReferenceFailure{_} _: Relation.Failed{}
    case Spec.ReferenceResult{state, reference}
      Core.Machine{facts, path, frames, _, _, _, nodes, slots, _}:
      Continuation.after_flat(
        Spec.flat_head(reference, owner, index, Spec.flatten(fuel)),
        state,
        Spec.Flat{
          List.reverse(&2, Core.PlannedNode, nodes),
          List.reverse(&2, Core.ReferenceSlot, slots)
        },
        Continuation.resume(frames, facts, path, fuel)
      )

def slot_exact(+fuel: Nat, +machine: Core.Machine, +slot: Core.ReferenceSlot) -> {
  Machine.evaluate(fuel, CoreTraversal.add_reference(machine, slot))
    == Machine.resume_prefix(
      State.frames(machine),
      CoreTraversal.machine_facts(machine),
      CoreTraversal.machine_path(machine),
      fuel,
      State.state(machine),
      Spec.flat_join(Prefix.prefix(machine), Spec.Flat{[], slot <> []})
    )
    : Relation.Observation
}:
  match machine:
    case Core.Machine{
      +facts,
      +path,
      +frames,
      +visited,
      +budget,
      +address,
      +nodes,
      +slots,
      +pending
    }:
      Equal.cong(
        Spec.Flat,
        Relation.Observation,
        Machine.resume_prefix(
          frames,
          facts,
          path,
          fuel,
          Spec.State{visited, budget, address, pending}
        ),
        Prefix.prefix(
          CoreTraversal.add_reference(
            Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, pending},
            slot
          )
        ),
        Spec.flat_join(
          Prefix.prefix(
            Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, pending}
          ),
          Spec.Flat{[], slot <> []}
        ),
        Prefix.add_reference_prefix(
          Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, pending},
          slot
        )
      )

def omitted_exact(+fuel: Nat, +machine: Core.Machine, +owner: Nat, +index: Nat,
    +symbol: String, +reason: Core.Omission) -> {
  Machine.evaluate(fuel, CoreTraversal.omit(machine, owner, index, symbol, reason))
    == observation(Spec.omitted(State.state(machine), symbol, reason), machine, fuel, owner, index)
    : Relation.Observation
}:
  match machine:
    case Core.Machine{
      +facts,
      +path,
      +frames,
      +visited,
      +budget,
      +address,
      +nodes,
      +slots,
      +pending
    }:
      slot_exact(
        fuel,
        Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, pending},
        Core.ReferenceSlot{owner, index, symbol, Core.Omitted{reason, symbol}}
      )

def included_exact(+fuel: Nat, +machine: Core.Machine, +owner: Nat, +index: Nat,
    +symbol: String, +identity: String) -> {
      Machine.evaluate(fuel, CoreTraversal.include(machine, owner, index, symbol, identity))
      == observation(
        Spec.ReferenceResult{
          State.state(machine),
          Spec.TreeReference{symbol, Spec.Included{identity}}
        },
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match machine:
    case Core.Machine{
      +facts,
      +path,
      +frames,
      +visited,
      +budget,
      +address,
      +nodes,
      +slots,
      +pending
    }:
      slot_exact(
        fuel,
        Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, pending},
        Core.ReferenceSlot{owner, index, symbol, Core.Included{identity}}
      )

def pending_omitted_exact(+fuel: Nat, +machine: Core.Machine, +owner: Nat, +index: Nat,
    +symbol: String, +reason: Core.Omission, +item: Core.Pending) -> {
  Machine.evaluate(fuel, CoreTraversal.add_pending(CoreTraversal.omit(machine, owner, index, symbol, reason), item))
    == observation(Spec.omitted(Spec.pending(State.state(machine), item), symbol, reason),
      machine, fuel, owner, index) : Relation.Observation
}:
  match machine:
    case Core.Machine{
      +facts,
      +path,
      +frames,
      +visited,
      +budget,
      +address,
      +nodes,
      +slots,
      +pending
    }:
      omitted_exact(
        fuel,
        Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, item <> pending},
        owner,
        index,
        symbol,
        reason
      )

def queue_import_exact(+fuel: Nat, +machine: Core.Machine, +owner: Nat, +index: Nat,
    +symbol: String, +depth: Nat, +expected: Core.ExpectedKind, +imported: Core.ImportBinding) -> {
      Machine.evaluate(
        fuel,
        CoreTraversal.queue_import(machine, owner, index, symbol, depth, expected, imported)
      )
      == observation(
        Spec.queue(
          State.state(machine),
          CoreTraversal.machine_path(machine),
          symbol,
          owner,
          index,
          depth,
          expected,
          imported
        ),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match imported:
    case Core.ImportBinding{+path, +name, _}:
      pending_omitted_exact(
        fuel,
        machine,
        owner,
        index,
        symbol,
        Core.Unavailable{},
        Core.Pending{
          owner,
          index,
          CoreTraversal.machine_path(machine),
          symbol,
          name,
          depth,
          expected,
          Core.Imported{path, name}
        }
      )

def import_kind_exact(+fuel: Nat, +expected: Core.ExpectedKind, +type_only: Bool,
    +machine: Core.Machine, +owner: Nat, +index: Nat, +symbol: String, +depth: Nat,
    +imported: Core.ImportBinding) -> {
      Machine.evaluate(
        fuel,
        CoreTraversal.import_by_kind(expected, type_only, machine, owner, index, symbol, depth, imported)
      )
      == observation(
        Spec.imported_kind(
          expected,
          type_only,
          State.state(machine),
          CoreTraversal.machine_path(machine),
          symbol,
          owner,
          index,
          depth,
          imported
        ),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match expected type_only:
    case Core.AnyKind{} True{}: queue_import_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.AnyKind{},
      imported
    )
    case Core.AnyKind{} False{}: queue_import_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.AnyKind{},
      imported
    )
    case Core.TypeKind{} True{}: queue_import_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.TypeKind{},
      imported
    )
    case Core.TypeKind{} False{}: queue_import_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.TypeKind{},
      imported
    )
    case Core.FunctionKind{} True{}: omitted_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      Core.UnsupportedTarget{}
    )
    case Core.FunctionKind{} False{}: queue_import_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.FunctionKind{},
      imported
    )

def import_present_exact(
  +fuel: Nat,
  +machine: Core.Machine,
  +owner: Nat,
  +index: Nat,
  +symbol: String,
  +depth: Nat,
  +expected: Core.ExpectedKind,
  +imported: Maybe<&2, Core.ImportBinding>
) -> {
      Machine.evaluate(
        fuel,
        CoreTraversal.import_present(machine, owner, index, symbol, depth, expected, imported)
      )
      == observation(
        Spec.unresolved_or_import(
          imported,
          State.state(machine),
          CoreTraversal.machine_path(machine),
          symbol,
          owner,
          index,
          depth,
          expected
        ),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match imported:
    case None{}: omitted_exact(fuel, machine, owner, index, symbol, Core.Unresolved{})
    case Some{Core.ImportBinding{+path, +name, +type_only}}:
      import_kind_exact(fuel, expected, type_only, machine, owner, index, symbol, depth,
        Core.ImportBinding{path, name, type_only})

def declaration_handle(declaration: Core.Declaration) -> Nat:
  match declaration:
    case Core.Declaration{handle, _, _, _, _}: handle

def bundled_queue_exact(+fuel: Nat, +machine: Core.Machine, +owner: Nat, +index: Nat,
    +symbol: String, +depth: Nat, +expected: Core.ExpectedKind, +declaration: Core.Declaration) -> {
      Machine.evaluate(
        fuel,
        CoreTraversal.bundled_queue(machine, owner, index, symbol, depth, expected, declaration)
      )
      == observation(
        Spec.bundle_queue(
          State.state(machine),
          CoreTraversal.machine_path(machine),
          symbol,
          owner,
          index,
          depth,
          expected,
          declaration_handle(declaration)
        ),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match declaration:
    case Core.Declaration{+handle, _, _, _, _}:
      pending_omitted_exact(
        fuel,
        machine,
        owner,
        index,
        symbol,
        Core.Unavailable{},
        Core.Pending{
          owner,
          index,
          CoreTraversal.machine_path(machine),
          symbol,
          symbol,
          depth,
          expected,
          Core.BundledTarget{handle}
        }
      )

def bundle_finish_exact(+fuel: Nat, +permit: Bool, +seen: Bool, +machine: Core.Machine,
    +declaration: Core.Declaration, +owner: Nat, +index: Nat, +symbol: String,
    +depth: Nat, +expected: Core.ExpectedKind) -> {
      Machine.evaluate(
        fuel,
        CoreTraversal.bundled_budget(
          permit,
          seen,
          machine,
          declaration,
          owner,
          index,
          symbol,
          depth,
          expected
        )
      )
      == observation(
        Spec.bundle_finish(
          permit,
          seen,
          State.state(machine),
          CoreTraversal.machine_path(machine),
          symbol,
          owner,
          index,
          depth,
          expected,
          declaration_handle(declaration),
          CoreTraversal.declaration_identity(declaration)
        ),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match permit seen:
    case False{} False{}: omitted_exact(fuel, machine, owner, index, symbol, Core.ReferenceLimit{})
    case False{} True{}: omitted_exact(fuel, machine, owner, index, symbol, Core.ReferenceLimit{})
    case True{} False{}: bundled_queue_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      depth,
      expected,
      declaration
    )
    case True{} True{}: included_exact(
      fuel,
      machine,
      owner,
      index,
      symbol,
      CoreTraversal.declaration_identity(declaration)
    )

def observe_result(machine: Core.Machine, fuel: Nat, owner: Nat, index: Nat,
    result: Spec.ReferenceResult) -> Relation.Observation:
  observation(result, machine, fuel, owner, index)

def observed_bundle_permit(
  state: Spec.State,
  path: String,
  symbol: String,
  +owner: Nat,
  +index: Nat,
  depth: Nat,
  expected: Core.ExpectedKind,
  handle: Nat,
  identity: String,
  seen: Bool,
  machine: Core.Machine,
  fuel: Nat,
  permit: Bool
) -> Relation.Observation:
  observation(Spec.bundle_finish(permit, seen, state, path, symbol, owner, index, depth,
    expected, handle, identity), machine, fuel, owner, index)

def bundle_charged_exact(+fuel: Nat, +machine: Core.Machine, +seen: Bool,
    +declaration: Core.Declaration, +owner: Nat, +index: Nat, +symbol: String,
    +depth: Nat, +expected: Core.ExpectedKind) -> {
      Machine.evaluate(
        fuel,
        CoreTraversal.bundled_charged(machine, seen, declaration, owner, index, symbol, depth, expected)
      )
      == observation(
        Spec.bundle_budget(
          State.state(machine),
          seen,
          CoreTraversal.machine_path(machine),
          symbol,
          owner,
          index,
          depth,
          expected,
          declaration_handle(declaration),
          CoreTraversal.declaration_identity(declaration)
        ),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  Equal.trans(
    Relation.Observation,
    Machine.evaluate(
      fuel,
      CoreTraversal.bundled_charged(machine, seen, declaration, owner, index, symbol, depth, expected)
    ),
    observed_bundle_permit(
      State.state(machine),
      CoreTraversal.machine_path(machine),
      symbol,
      owner,
      index,
      depth,
      expected,
      declaration_handle(declaration),
      CoreTraversal.declaration_identity(declaration),
      seen,
      machine,
      fuel,
      CoreTraversal.machine_allows(machine)
    ),
    observation(
      Spec.bundle_budget(
        State.state(machine),
        seen,
        CoreTraversal.machine_path(machine),
        symbol,
        owner,
        index,
        depth,
        expected,
        declaration_handle(declaration),
        CoreTraversal.declaration_identity(declaration)
      ),
      machine,
      fuel,
      owner,
      index
    ),
    bundle_finish_exact(
      fuel,
      CoreTraversal.machine_allows(machine),
      seen,
      machine,
      declaration,
      owner,
      index,
      symbol,
      depth,
      expected
    ),
    Equal.cong(
      Bool,
      Relation.Observation,
      observed_bundle_permit(
        State.state(machine),
        CoreTraversal.machine_path(machine),
        symbol,
        owner,
        index,
        depth,
        expected,
        declaration_handle(declaration),
        CoreTraversal.declaration_identity(declaration),
        seen,
        machine,
        fuel
      ),
      CoreTraversal.machine_allows(machine),
      Spec.allowed(State.state(machine), CoreTraversal.machine_path(machine)),
      State.allowed_exact(machine)
    )
  )

# Context ignores shared state because the actual reference result already owns it.
def budget_context(+result: Spec.ReferenceResult, +machine: Core.Machine, budget: Core.Budget,
    +fuel: Nat, +owner: Nat, +index: Nat) -> {
  observation(result, CoreTraversal.replace_budget(machine, budget), fuel, owner, index)
    == observation(result, machine, fuel, owner, index) : Relation.Observation
}:
  match result machine:
    case Spec.ReferenceFailure{_} Core.Machine{_, _, _, _, _, _, _, _, _}: {==}
    case Spec.ReferenceResult{_, _} Core.Machine{_, _, _, _, _, _, _, _, _}: {==}

def reference_next_exact(+result: Spec.ReferenceResult, +machine: Core.Machine,
    +fuel: Nat, +owner: Nat, +index: Nat) -> {
      Child.reference_next(
        result,
        fuel,
        owner,
        index,
        Prefix.prefix(machine),
        Continuation.resume(
          State.frames(machine),
          CoreTraversal.machine_facts(machine),
          CoreTraversal.machine_path(machine),
          fuel
        )
      )
      == observation(result, machine, fuel, owner, index)
      : Relation.Observation
    }:
  match result machine:
    case Spec.ReferenceFailure{_} Core.Machine{_, _, _, _, _, _, _, _, _}: {==}
    case Spec.ReferenceResult{_, _} Core.Machine{_, _, _, _, _, _, _, _, _}: {==}

def selected_expansion_exact(
  +fuel: Nat,
  +machine: Core.Machine,
  +declaration: Core.Declaration,
  +owner: Nat,
  +index: Nat,
  +symbol: String,
  +expected: Core.ExpectedKind,
  +depth: Nat,
  +valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool},
  +enough: {Nat.is_le(Relation.spec_bound(CoreTraversal.machine_facts(machine)), fuel) == True{} : Bool},
  found: {
    Spec.find_declaration(CoreTraversal.machine_facts(machine), symbol, expected)
    == Some{declaration}
    : Maybe<&2, Core.Declaration>
  }
) -> {
      Machine.evaluate(fuel, CoreTraversal.expand_node(machine, declaration, owner, index, symbol, depth))
      == observation(
        Spec.expanded_result(Child.root_result(fuel, machine, symbol, expected, depth), symbol),
        machine,
        fuel,
        owner,
        index
      )
      : Relation.Observation
    }:
  match machine:
    case Core.Machine{
      +facts,
      +path,
      +frames,
      +visited,
      +budget,
      +address,
      +nodes,
      +slots,
      +pending
    }:
      +machine = {Core.Machine{facts, path, frames, visited, budget, address, nodes, slots, pending} : Core.Machine}
      Equal.trans(
        Relation.Observation,
        Machine.evaluate(fuel, CoreTraversal.expand_node(machine, declaration, owner, index, symbol, depth)),
        Child.reference_next(
          Spec.expanded_result(
            Child.root_result((1n + fuel : Nat), machine, symbol, expected, depth),
            symbol
          ),
          (1n + fuel : Nat),
          owner,
          index,
          Prefix.prefix(machine),
          Continuation.resume(frames, facts, path, fuel)
        ),
        observation(
          Spec.expanded_result(Child.root_result(fuel, machine, symbol, expected, depth), symbol),
          machine,
          fuel,
          owner,
          index
        ),
        Machine.expanded_evaluation(
          fuel,
          machine,
          declaration,
          owner,
          index,
          symbol,
          expected,
          depth,
          found
        ),
        Equal.trans(
          Relation.Observation,
          Child.reference_next(
            Spec.expanded_result(
              Child.root_result((1n + fuel : Nat), machine, symbol, expected, depth),
              symbol
            ),
            (1n + fuel : Nat),
            owner,
            index,
            Prefix.prefix(machine),
            Continuation.resume(frames, facts, path, fuel)
          ),
          Child.reference_next(
            Spec.expanded_result(Child.root_result(fuel, machine, symbol, expected, depth), symbol),
            fuel,
            owner,
            index,
            Prefix.prefix(machine),
            Continuation.resume(frames, facts, path, fuel)
          ),
          observation(
            Spec.expanded_result(Child.root_result(fuel, machine, symbol, expected, depth), symbol),
            machine,
            fuel,
            owner,
            index
          ),
          Expansion.root_fuels_equal(
            (1n + fuel : Nat),
            fuel,
            machine,
            symbol,
            expected,
            depth,
            owner,
            index,
            Prefix.prefix(machine),
            Expansion.root_complete(
              (1n + fuel : Nat),
              machine,
              symbol,
              expected,
              depth,
              valid,
              Order.le_trans(
                Relation.spec_bound(facts),
                fuel,
                (1n + fuel : Nat),
                enough,
                NatFacts.le_succ(fuel)
              )
            ),
            Expansion.root_complete(fuel, machine, symbol, expected, depth, valid, enough),
            Continuation.resume(frames, facts, path, fuel)
          ),
          reference_next_exact(
            Spec.expanded_result(Child.root_result(fuel, machine, symbol, expected, depth), symbol),
            machine,
            fuel,
            owner,
            index
          )
        )
      )
