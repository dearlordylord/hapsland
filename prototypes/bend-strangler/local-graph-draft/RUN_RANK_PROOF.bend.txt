# Internal transports for the approved adequate-fuel theorem, on actual String machines.
def terminal_type(machine: Core.Machine) -> Data:
  {CoreTraversal.done(machine) == True{} : Bool}

def zero_rank_terminal(+machine: Core.Machine,
  enough: {Nat.is_le(Rank.rank(machine), 0n) == True{} : Bool}) -> terminal_type(machine):
  match machine:
    case Core.Machine{_, _, [], _, _, _, _, _, _}: {==}
    case Core.Machine{
      +facts,
      +path,
      +frame <> +tail,
      +visited,
      +budget,
      +next,
      +nodes,
      +slots,
      +pending
    }:
      +original = {Core.Machine{facts, path, frame <> tail, visited, budget, next, nodes, slots, pending} : Core.Machine}
      +base = (2n * Rank.unseen_catalog(visited, Rank.declaration_catalog(facts)) + 2n * Rank.pending_references(
        frame <> tail
      ) : Nat)
      positive : {Nat.is_lt(0n, Rank.rank(original)) == True{} : Bool} =
        %NatFacts.add_comm((1n + List.length(&2, Core.Frame, tail) : Nat), base):
          {Nat.is_lt(0n, _) == True{} : Bool}
        {==}
      not_bound = NatFacts.not_le_of_lt(0n, Rank.rank(original), positive)
      impossible = BoolFacts.internal_false_ne_true(
        Equal.trans(Bool, False{}, Nat.is_le(Rank.rank(original), 0n), True{},
          Equal.sym(Bool, Nat.is_le(Rank.rank(original), 0n), False{}, not_bound), enough))
      Empty.absurd(terminal_type(original), impossible)

def zero_fuel_completes(+machine: Core.Machine,
  enough: {Nat.is_le(Rank.rank(machine), 0n) == True{} : Bool}) -> {
  Rank.successful(CoreTraversal.run(0n, machine)) == True{} : Bool
}:
  %Equal.sym(Bool, CoreTraversal.done(machine), True{}, zero_rank_terminal(machine, enough)):
    {Rank.successful(CoreTraversal.run_zero(_, machine)) == True{} : Bool}
  {==}

def continuation_type(+fuel: Nat) -> Type:
  @+next: Core.Machine ->
    {Patricia.canonical(Bool, CoreTraversal.machine_visited(next)) == True{} : Bool} ->
    {Nat.is_le(Rank.rank(next), fuel) == True{} : Bool} ->
    {Rank.successful(CoreTraversal.run(fuel, next)) == True{} : Bool}

def step_fuel_bound(
  +machine: Core.Machine,
  +fuel: Nat,
  decrease: {
    Nat.is_le((Rank.rank(CoreTraversal.step(machine)) + 1n : Nat), Rank.rank(machine))
    == True{}
    : Bool
  },
  enough: {Nat.is_le(Rank.rank(machine), (fuel + 1n : Nat)) == True{} : Bool}
) -> {
  Nat.is_le(Rank.rank(CoreTraversal.step(machine)), fuel) == True{} : Bool
}:
  %NatFacts.add_le_add_iff_right_sym(Rank.rank(CoreTraversal.step(machine)), fuel, 1n):
    {_ == True{} : Bool}
  NatFacts.le_trans(Nat.add(Rank.rank(CoreTraversal.step(machine)), 1n), Rank.rank(machine),
    Nat.add(fuel, 1n), decrease, enough)

def run_if_completes(+complete: Bool, +machine: Core.Machine, +fuel: Nat,
  valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool},
  agreement: {CoreTraversal.done(machine) == complete : Bool},
  enough: {Nat.is_le(Rank.rank(machine), (fuel + 1n : Nat)) == True{} : Bool},
  suffix: continuation_type(fuel)) -> {
  Rank.successful(CoreTraversal.run_if(complete, machine, next => CoreTraversal.run(fuel, next))) == True{} : Bool
}:
  match complete:
    case True{}: {==}
    case False{}:
      +bounded = Step.step_decreases_preserving_visited(machine, valid, agreement)
      lawful = BoolFacts.eq_true_of_and_left(
        Patricia.canonical(Bool, CoreTraversal.machine_visited(CoreTraversal.step(machine))),
        Nat.is_le((Rank.rank(CoreTraversal.step(machine)) + 1n : Nat), Rank.rank(machine)), bounded)
      decrease = BoolFacts.eq_true_of_and_right(
        Patricia.canonical(Bool, CoreTraversal.machine_visited(CoreTraversal.step(machine))),
        Nat.is_le((Rank.rank(CoreTraversal.step(machine)) + 1n : Nat), Rank.rank(machine)), bounded)
      suffix(CoreTraversal.step(machine), lawful, step_fuel_bound(machine, fuel, decrease, enough))

def rank_le(machine: Core.Machine, fuel: Nat) -> Bool:
  Nat.is_le(Rank.rank(machine), fuel)

law run_completes_from_rank:
  for +fuel: Nat
  for +machine: Core.Machine
  for valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool}
  for enough: {Nat.is_le(Rank.rank(machine), fuel) == True{} : Bool}
  {Rank.successful(CoreTraversal.run(fuel, machine)) == True{} : Bool}

def run_completes_from_rank(fuel, machine, valid, enough):
  match fuel:
    case 0n: zero_fuel_completes(machine, enough)
    case 1n++rest:
      continue = {+next => lawful => bound => run_completes_from_rank(rest, next, lawful, bound) : continuation_type(rest)}
      adjusted = Equal.trans(
        Bool,
        rank_le(machine, Nat.add(rest, 1n)),
        rank_le(machine, Nat.add(1n, rest)),
        True{},
        Equal.cong(
          Nat,
          Bool,
          rank_le(machine),
          Nat.add(rest, 1n),
          Nat.add(1n, rest),
          NatFacts.add_comm(rest, 1n)
        ),
        enough
      )
      run_if_completes(CoreTraversal.done(machine), machine, rest, valid, {==}, adjusted, continue)
