def contract(+fuel: Nat, +catalog: Map<&2, Maybe<&2, Core.Declaration>>) -> Type:
  @+request: Spec.Request ->
    {Patricia.canonical(Bool, Measure.visited(Measure.request_state(request)))==True{}:Bool} ->
    {
      R.holds(
        Measure.visited(Measure.result_state(Spec.interpret(fuel, request))),
        Measure.visited(Measure.request_state(request)),
        catalog
      )
      ==True{}
      :Bool
    }

def child_or_cycle_progress(
  +fuel: Nat,
  +already_seen: Bool,
  +state: Spec.State,
  +facts: Core.Facts,
  +path: String,
  +symbol: String,
  +depth: Nat,
  +expected: Core.ExpectedKind,
  +identity: String,
  +before: Map<&2, Bool>,
  +catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +incoming: {R.holds(Measure.visited(state), before, catalog)==True{}:Bool},
  child: contract(fuel, catalog)
) -> {
    R.holds(
      Prim.reference_visited(
        Spec.child_or_cycle(
          already_seen,
          state,
          facts,
          path,
          symbol,
          depth,
          expected,
          identity,
          C.build_child(fuel)
        )
      ),
      before,
      catalog
    )
    ==True{}
    :Bool
  }:
  match already_seen:
    case True{}: incoming
    case False{}:
      +request={Spec.Request{facts, path, symbol, expected, (depth+1n:Nat), state} : Spec.Request}
      +result=Spec.interpret(fuel, request)
      after=child(request, R.lawful(Measure.visited(state), before, catalog, incoming))
      total=R.compose(
        Measure.visited(Measure.result_state(result)),
        Measure.visited(state),
        before,
        catalog,
        after,
        incoming
      )
      R.replace_after(
        Prim.reference_visited(Spec.expanded_result(Spec.as_build(result), symbol)),
        Measure.visited(Measure.result_state(result)),
        before,
        catalog,
        Prim.expanded_dispatch_visited_exact(result, symbol),
        total
      )

def references_child(
  fuel: Nat,
  items: List<&2, Core.Reference>,
  facts: Core.Facts,
  path: String,
  owner: Nat,
  index: Nat,
  depth: Nat,
  state: Spec.State
) -> Spec.ReferencesResult:
  Spec.as_references(
    Spec.interpret(fuel, Spec.ReferencesRequest{items, facts, path, owner, index, depth, state})
  )

def after_reference_progress(
  +fuel: Nat,
  +result: Spec.ReferenceResult,
  +items: List<&2, Core.Reference>,
  +facts: Core.Facts,
  +path: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +before: Map<&2, Bool>,
  +catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +incoming: {R.holds(Prim.reference_visited(result), before, catalog)==True{}:Bool},
  sibling: contract(fuel, catalog)
) -> {
    R.holds(
      Measure.visited(
        Measure.references_state(
          Spec.after_reference(
            result,
            references_child(fuel, items, facts, path, owner, index, depth)
          )
        )
      ),
      before,
      catalog
    )
    ==True{}
    :Bool
  }:
  match result:
    case Spec.ReferenceFailure{state}: incoming
    case Spec.ReferenceResult{+state, +reference}:
      +request={Spec.ReferencesRequest{items, facts, path, owner, index, depth, state} : Spec.Request}
      +answer=Spec.interpret(fuel, request)
      progress=sibling(request, R.lawful(Measure.visited(state), before, catalog, incoming))
      total=R.compose(
        Measure.visited(Measure.result_state(answer)),
        Measure.visited(state),
        before,
        catalog,
        progress,
        incoming
      )
      R.replace_after(
        Measure.visited(
          Measure.references_state(Spec.prepend(Spec.as_references(answer), reference))
        ),
        Measure.visited(Measure.result_state(answer)),
        before,
        catalog,
        Prim.prepend_dispatch_visited_exact(answer, reference),
        total
      )

def references_progress(
  +fuel: Nat,
  +items: List<&2, Core.Reference>,
  +facts: Core.Facts,
  +path: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +state: Spec.State,
  +before: Map<&2, Bool>,
  +catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +incoming: {R.holds(Measure.visited(state), before, catalog)==True{}:Bool},
  child: contract(fuel, catalog),
  sibling: contract(fuel, catalog)
) -> {
    R.holds(
      Measure.visited(
        Measure.references_state(
          Spec.references(
            items,
            facts,
            path,
            owner,
            index,
            depth,
            state,
            Spec.interpret(fuel),
            Spec.interpret(fuel)
          )
        )
      ),
      before,
      catalog
    )
    ==True{}
    :Bool
  }:
  match items:
    case []: incoming
    case +head <> +tail:
      first=reference_progress(
        fuel,
        head,
        facts,
        path,
        owner,
        index,
        depth,
        state,
        before,
        catalog,
        incoming,
        child
      )
      after_reference_progress(
        fuel,
        Spec.reference(head, facts, path, owner, index, depth, state, C.build_child(fuel)),
        tail,
        facts,
        path,
        owner,
        (index+1n:Nat),
        depth,
        before,
        catalog,
        first,
        sibling
      )

def walk_progress(
  enter: R.enter_type(),
  +fuel: Nat,
  +found: Maybe<&2, Core.Declaration>,
  +facts: Core.Facts,
  +path: String,
  +state: Spec.State,
  +depth: Nat,
  +before: Map<&2, Bool>,
  +catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  +incoming: {R.holds(Measure.visited(state), before, catalog)==True{}:Bool},
  child: contract(fuel, catalog)
) -> {
    R.holds(
      Measure.visited(
        Measure.build_state(
          Spec.walk_declaration(found, facts, path, state, depth, Spec.interpret(fuel))
        )
      ),
      before,
      catalog
    )
    ==True{}
    :Bool
  }:
  match found:
    case None{}: incoming
    case Some{Core.Declaration{+artifact, +identity, _, _, +children}}:
      +entered=Spec.enter(state, path, identity)
      +entered_progress=enter(state, path, identity, before, catalog, incoming)
      +request={Spec.ReferencesRequest{children, facts, path, Spec.address(state), 0n, depth, entered} : Spec.Request}
      +answer=Spec.interpret(fuel, request)
      progress=child(request, R.lawful(Measure.visited(entered), before, catalog, entered_progress))
      total=R.compose(
        Measure.visited(Measure.result_state(answer)),
        Measure.visited(entered),
        before,
        catalog,
        progress,
        entered_progress
      )
      R.replace_after(
        Measure.visited(
          Measure.build_state(
            Spec.tree_result(Spec.as_references(answer), Spec.address(state), artifact)
          )
        ),
        Measure.visited(Measure.result_state(answer)),
        before,
        catalog,
        Prim.tree_dispatch_visited_exact(answer, Spec.address(state), artifact),
        total
      )

def dispatch_progress(
  enter: R.enter_type(),
  +fuel: Nat,
  +request: Spec.Request,
  +catalog: Map<&2, Maybe<&2, Core.Declaration>>,
  valid: {Patricia.canonical(Bool, Measure.visited(Measure.request_state(request)))==True{}:Bool},
  child: contract(fuel, catalog),
  sibling: contract(fuel, catalog)
) -> {
    R.holds(
      Measure.visited(
        Measure.result_state(Spec.dispatch(request, Spec.interpret(fuel), Spec.interpret(fuel)))
      ),
      Measure.visited(Measure.request_state(request)),
      catalog
    )
    ==True{}
    :Bool
  }:
  match request:
    case Spec.Request{+facts, +path, +name, +expected, +depth, +state}:
      walk_progress(
        enter,
        fuel,
        Spec.find_declaration(facts, name, expected),
        facts,
        path,
        state,
        depth,
        Measure.visited(state),
        catalog,
        R.reflexive(Measure.visited(state), catalog, valid),
        child
      )
    case Spec.ReferencesRequest{+items, +facts, +path, +owner, +index, +depth, +state}:
      references_progress(
        fuel,
        items,
        facts,
        path,
        owner,
        index,
        depth,
        state,
        Measure.visited(state),
        catalog,
        R.reflexive(Measure.visited(state), catalog, valid),
        child,
        sibling
      )
