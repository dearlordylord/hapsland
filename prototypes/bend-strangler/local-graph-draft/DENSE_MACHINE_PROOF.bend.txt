# Internal proof invariant for the actual reverse allocation accumulator.
# Neither map validity nor successful termination is assumed.
def backward(nodes: List<&2, Core.PlannedNode>, next: Nat) -> Bool:
  match nodes next:
    case [] 0n: True{}
    case Core.PlannedNode{node, _} <> rest 1n++previous:
      Bool.and(Nat.is_eq(node, previous), backward(rest, previous))
    case _ _: False{}

def allocation_valid(machine: Core.Machine) -> Bool:
  match machine:
    case Core.Machine{_, _, _, _, _, next, +nodes, _, _}:
      Bool.and(backward(nodes, next), Dense.nonempty(nodes))

def successor(+n: Nat) -> {(n + 1n : Nat) == 1n+n : Nat}:
  N.add_comm(n, 1n)

def enter_node_valid(+machine: Core.Machine, +declaration: Core.Declaration, +depth: Nat,
    valid: {allocation_valid(machine) == True{} : Bool}) -> {
  allocation_valid(CoreTraversal.enter_node(machine, declaration, depth)) == True{} : Bool
}:
  match machine declaration:
    case Core.Machine{facts, path, frames, visited, budget, +next, +nodes, refs, pending}
        Core.Declaration{+artifact, identity, function, bundled, children}:
      %Equal.sym(Nat, (next + 1n : Nat), 1n+next, successor(next)):
        {Bool.and(backward(Core.PlannedNode{next, artifact} <> nodes, _), True{}) == True{} : Bool}
      B.and_eq_true(
        Bool.and(Nat.is_eq(next, next), backward(nodes, next)), True{},
        B.and_eq_true(Nat.is_eq(next, next), backward(nodes, next), NatEquality.is_eq_refl(next),
          B.eq_true_of_and_left(backward(nodes, next), Dense.nonempty(nodes), valid)), {==})

def expand_node_valid(
  +machine: Core.Machine,
  +declaration: Core.Declaration,
  +owner: Nat,
  +index: Nat,
  +name: String,

  valid: {allocation_valid(machine) == True{} : Bool}
) -> {
      allocation_valid(CoreTraversal.expand_node(machine, declaration, owner, index, name, depth))
      == True{}
      : Bool
    }:

    CoreTraversal.add_reference(
      machine,

    add_reference_valid(
      machine,
      Core.ReferenceSlot{owner, index, name, Core.Expanded{CoreTraversal.next_node(machine)}},
      valid
    )
  )

def queue_import_valid(
  +machine: Core.Machine,
  +owner: Nat,
  +index: Nat,
  +symbol: String,
  +depth: Nat,
  +expected: Core.ExpectedKind,
  +imported: Core.ImportBinding,
  valid: {allocation_valid(machine) == True{} : Bool}
) -> {
      allocation_valid(CoreTraversal.queue_import(machine, owner, index, symbol, depth, expected, imported))
      == True{}
      : Bool
    }:
  match imported:
    case Core.ImportBinding{+path, +name, _}: add_pending_valid(
      CoreTraversal.omit(machine, owner, index, symbol, Core.Unavailable{}),
      Core.Pending{
        owner,
        index,
        CoreTraversal.machine_path(machine),
        symbol,
        name,
        depth,
        expected,
        Core.Imported{path, name}
      },
      omit_valid(machine, owner, index, symbol, Core.Unavailable{}, valid)
    )

def import_by_kind_valid(
  +expected: Core.ExpectedKind,
  +type_only: Bool,
  +machine: Core.Machine,
  +owner: Nat,
  +index: Nat,
  +symbol: String,
  +depth: Nat,
  +imported: Core.ImportBinding,
  valid: {allocation_valid(machine) == True{} : Bool}
) -> {
      allocation_valid(
        CoreTraversal.import_by_kind(expected, type_only, machine, owner, index, symbol, depth, imported)
      )
      == True{}
      : Bool
    }:
  match expected type_only:
    case Core.FunctionKind{} True{}: omit_valid(
      machine,
      owner,
      index,
      symbol,
      Core.UnsupportedTarget{},
      valid
    )
    case Core.FunctionKind{} False{}: queue_import_valid(
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.FunctionKind{},
      imported,
      valid
    )
    case Core.AnyKind{} True{}: queue_import_valid(
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.AnyKind{},
      imported,
      valid
    )
    case Core.AnyKind{} False{}: queue_import_valid(
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.AnyKind{},
      imported,
      valid
    )
    case Core.TypeKind{} True{}: queue_import_valid(
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.TypeKind{},
      imported,
      valid
    )
    case Core.TypeKind{} False{}: queue_import_valid(
      machine,
      owner,
      index,
      symbol,
      depth,
      Core.TypeKind{},
      imported,
      valid
    )

      machine,
      owner,
      index,
      symbol,
      depth,
      expected,
      valid
    )

def visit_reference_valid(
  +machine: Core.Machine,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +reference: Core.Reference,
  valid: {allocation_valid(machine) == True{} : Bool}
) -> {
  allocation_valid(CoreTraversal.visit_reference(machine, owner, index, depth, reference)) == True{} : Bool
}:
  match reference:
    case Core.Reference{+kind, +name, +expected, +target}: reference_kind_valid(
      kind,
      target,
      CoreTraversal.observe_machine_depth(machine, (depth + 1n : Nat)),
      owner,
      index,
      name,
      depth,
      expected,
      observe_machine_depth_valid(machine, (depth + 1n : Nat), valid)
    )

def push_frame_valid(+machine: Core.Machine, +frame: Core.Frame,
    valid: {allocation_valid(machine) == True{} : Bool}) -> {
  allocation_valid(CoreTraversal.push_frame(machine, frame)) == True{} : Bool
}:

    case Core.Machine{
      +facts,
      +path,
      +frames,
      +visited,
      +budget,
      +next,
      +nodes,
      +references,
      +pending
    }: valid

def process_frame_valid(+machine: Core.Machine, +frame: Core.Frame,
    valid: {allocation_valid(machine) == True{} : Bool}) -> {
  allocation_valid(CoreTraversal.process_frame(machine, frame)) == True{} : Bool
}:
  match frame:
    case Core.Frame{_, _, _, []}: valid
    case Core.Frame{+owner, +depth, +index, +reference <> +rest}: visit_reference_valid(
      CoreTraversal.push_frame(machine, Core.Frame{owner, depth, (index + 1n : Nat), rest}),
      owner,
      index,
      depth,
      reference,
      push_frame_valid(machine, Core.Frame{owner, depth, (index + 1n : Nat), rest}, valid)
    )

def step_valid(+machine: Core.Machine,
    valid: {allocation_valid(machine) == True{} : Bool}) -> {
  allocation_valid(CoreTraversal.step(machine)) == True{} : Bool
}:
  match machine:
    case Core.Machine{+facts, +path, [], +visited, +budget, +next, +nodes, +references, +pending}: valid
    case Core.Machine{
      +facts,
      +path,
      +frame <> +tail,
      +visited,
      +budget,
      +next,
      +nodes,
      +references,
      +pending
    }: process_frame_valid(
      Core.Machine{facts, path, tail, visited, budget, next, nodes, references, pending},
      frame,
      valid
    )
