# Actual child interpreter, fixed independently of the induction proof argument.
def selected_value_exact(-V: Data, pair: Map<&2, V> & V) -> {
  Core.map_result(V, pair) == Spec.value(V, pair) : V
}:
  match pair:
    case Tuple{_, value}: {==}

def seen_core_exact(+state: Spec.State, +identity: String) -> {
  CoreTraversal.has_identity(Measure.visited(state), identity) == Spec.seen(state, identity) : Bool
}:
  match state:
    case Spec.State{+visited, budget, address, pending}:
      selected_value_exact(Bool, Map.has(&2, Bool, visited, identity))

def valid_same(+after: Spec.State, +before: Spec.State,
    same: {Measure.visited(after) == Measure.visited(before) : Map<&2, Bool>},
    valid: {Patricia.canonical(Bool, Measure.visited(before)) == True{} : Bool}) -> {
  Patricia.canonical(Bool, Measure.visited(after)) == True{} : Bool
}:
  %Equal.sym(Map<&2, Bool>, Measure.visited(after), Measure.visited(before), same):
    {Patricia.canonical(Bool, _) == True{} : Bool}
  valid

def allowance_same(+facts: Core.Facts, +after: Spec.State, +before: Spec.State,
    +fuel: Nat,
    same: {Measure.visited(after) == Measure.visited(before) : Map<&2, Bool>},
    enough: {Nat.is_le(C.allowance(facts, before), fuel) == True{} : Bool}) -> {
  Nat.is_le(C.allowance(facts, after), fuel) == True{} : Bool
}:
  %Equal.sym(Map<&2, Bool>, Measure.visited(after), Measure.visited(before), same):
    {Nat.is_le((3n * Rank.unseen_catalog(_, Rank.declaration_catalog(facts)) + 3n : Nat), fuel)
      == True{} : Bool}
  enough

def root_dispatch_complete(+fuel: Nat, +facts: Core.Facts, +path: String,
    +symbol: String, +expected: Core.ExpectedKind, +depth: Nat, +state: Spec.State,
    +result: Spec.DispatchResult,
    complete: {Measure.projected_success(fuel,
      Spec.Request{facts, path, symbol, expected, depth, state}, result) == True{} : Bool}) -> {
  C.build_complete(fuel, Spec.as_build(result)) == True{} : Bool
}:
  match result:
    case Spec.BuildResult{Spec.Missing{state}}: complete
    case Spec.BuildResult{Spec.Built{state, tree}}: complete
    case Spec.BuildResult{Spec.InsufficientBound{state}}: complete
    case Spec.ListResult{Spec.ReferencesResult{state, references}}: complete
    case Spec.ListResult{Spec.ReferencesFailure{state}}: complete

def child_or_cycle_complete(
  selected_rank: C.selected_rank_type(),
  +fuel: Nat,
  +already_seen: Bool,
  +state: Spec.State,
  +facts: Core.Facts,
  +path: String,
  +symbol: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +expected: Core.ExpectedKind,
  +declaration: Core.Declaration,
  valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool},
  selected: {
    Spec.find_declaration(facts, symbol, expected)
    == Some{declaration}
    : Maybe<&2, Core.Declaration>
  },
  tested: {Spec.seen(state, CoreTraversal.declaration_identity(declaration)) == already_seen : Bool},
  enough: {Nat.is_le(C.allowance(facts, state), fuel) == True{} : Bool},
  child: C.contract(fuel)
) -> {
  C.reference_complete(fuel, owner, index,
    Spec.child_or_cycle(already_seen, state, facts, path, symbol, depth, expected,
      CoreTraversal.declaration_identity(declaration), C.build_child(fuel))) == True{} : Bool
}:
  match already_seen:
    case True{}: {==}
    case False{}:
      +valid = valid
      +request = {Spec.Request{facts, path, symbol, expected, (depth + 1n : Nat), state} : Spec.Request}
      +result = Spec.interpret(fuel, request)
      bound = selected_rank(
        facts,
        symbol,
        expected,
        Measure.visited(state),
        declaration,
        valid,
        selected,
        Equal.trans(
          Bool,
          CoreTraversal.has_identity(Measure.visited(state), CoreTraversal.declaration_identity(declaration)),
          Spec.seen(state, CoreTraversal.declaration_identity(declaration)),
          False{},
          seen_core_exact(state, CoreTraversal.declaration_identity(declaration)),
          tested
        )
      )
      complete = child(
        request,
        valid,
        Order.le_trans(
          Measure.request_rank(request),
          C.allowance(facts, state),
          fuel,
          bound,
          enough
        )
      )
      Leaf.expanded_complete(fuel, owner, index, symbol, Spec.as_build(result),
        root_dispatch_complete(fuel, facts, path, symbol, expected, (depth + 1n : Nat), state,
          result, complete))

    == Some{declaration}

  },

      CoreTraversal.declaration_identity(declaration), C.build_child(fuel))) == True{} : Bool
}:
  match permit:
    case False{}: {==}
    case True{}:
      child_or_cycle_complete(
        selected_rank,

        Spec.seen(state, CoreTraversal.declaration_identity(declaration)),
        state,
        facts,
        path,
        symbol,
        owner,
        index,

        expected,
        declaration,
        valid,
        selected,
        {==},
        enough,
        child
      )

def choose_local_complete(
  selected_rank: C.selected_rank_type(),
  +fuel: Nat,
  +local: Maybe<&2, Core.Declaration>,
  +imported: Maybe<&2, Core.ImportBinding>,
  +state: Spec.State,
  +facts: Core.Facts,
  +path: String,
  +symbol: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +expected: Core.ExpectedKind,
  valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool},
  selected: {Spec.find_declaration(facts, symbol, expected) == local : Maybe<&2, Core.Declaration>},
  enough: {Nat.is_le(C.allowance(facts, state), fuel) == True{} : Bool},
  child: C.contract(fuel)
) -> {
  C.reference_complete(fuel, owner, index,
    Spec.choose_local(local, imported, state, facts, path, symbol, owner, index,
      depth, expected, C.build_child(fuel))) == True{} : Bool
}:
  match local:
    case Some{declaration}:
      local_conflict_complete(
        selected_rank,
        fuel,
        imported,
        declaration,
        state,
        facts,
        path,
        symbol,
        owner,
        index,
        depth,
        expected,
        valid,
        selected,
        enough,
        child
      )
    case None{}:
      Leaf.leaf_complete(fuel, owner, index,
        Spec.import_budget(Spec.allowed(state, path), imported, state, path,
          symbol, owner, index, depth, expected),
        Leaf.import_budget_leaf(Spec.allowed(state, path), imported, state, path,
          symbol, owner, index, depth, expected))

def ordinary_found_complete(
  selected_rank: C.selected_rank_type(),
  +fuel: Nat,
  +local: Maybe<&2, Core.Declaration>,
  +imported: Maybe<&2, Core.ImportBinding>,
  +state: Spec.State,
  +facts: Core.Facts,
  +path: String,
  +symbol: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +expected: Core.ExpectedKind,
  valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool},
  selected: {Spec.find_declaration(facts, symbol, expected) == local : Maybe<&2, Core.Declaration>},
  enough: {Nat.is_le(C.allowance(facts, state), fuel) == True{} : Bool},
  child: C.contract(fuel)
) -> {
  C.reference_complete(fuel, owner, index,
    Spec.ordinary_found(local, imported, state, facts, path, symbol, owner, index,
      depth, expected, C.build_child(fuel))) == True{} : Bool
}:
  +identity = Spec.identity_key(local, imported, path, symbol)
  +next = Spec.target(state, path, identity)
  choose_local_complete(
    selected_rank,
    fuel,
    local,
    imported,
    next,
    facts,
    path,
    symbol,
    owner,
    index,
    depth,
    expected,
    valid_same(next, state, Prim.target_visited_exact(state, path, identity), valid),
    selected,
    allowance_same(
      facts,
      next,
      state,
      fuel,
      Prim.target_visited_exact(state, path, identity),
      enough
    ),
    child
  )

def reference_complete(
  selected_rank: C.selected_rank_type(),
  +fuel: Nat,
  +reference: Core.Reference,
  +facts: Core.Facts,
  +path: String,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +state: Spec.State,
  valid: {Patricia.canonical(Bool, Measure.visited(state)) == True{} : Bool},
  enough: {Nat.is_le(C.allowance(facts, state), fuel) == True{} : Bool},
  child: C.contract(fuel)
) -> {
  C.reference_complete(fuel, owner, index,
    Spec.reference(reference, facts, path, owner, index, depth, state,
      C.build_child(fuel))) == True{} : Bool
}:
  match reference:
    case Core.Reference{kind, symbol, expected, target_id}:
      +next = Spec.reference_depth(state, (depth + 1n : Nat))
      by_kind_complete(
        selected_rank,
        fuel,
        kind,
        target_id,
        facts,
        path,
        symbol,
        owner,
        index,
        depth,
        expected,
        next,
        valid_same(
          next,
          state,
          Prim.reference_depth_visited_exact(state, (depth + 1n : Nat)),
          valid
        ),
        allowance_same(
          facts,
          next,
          state,
          fuel,
          Prim.reference_depth_visited_exact(state, (depth + 1n : Nat)),
          enough
        ),
        child
      )
