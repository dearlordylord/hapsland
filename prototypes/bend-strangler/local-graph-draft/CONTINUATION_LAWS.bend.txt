import Base
import ./core.bend as Core
import ./Traversal.bend as CoreTraversal
import ../local-graph-representation-prototypes/list-state/PATRICIA_RELATION.bend as Patricia
import ./RANK.bend as Rank
import ./RELATION.bend as Relation
import ./CONTINUATION.bend as Continuation

# The whole independent suffix completes for arbitrary raw catalogs, frames and output prefixes.
# Only visited routing is canonical; no reachable-state/address/unique-identity premise is added.
law continuation_completes:
  for +machine: Core.Machine
  for valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool}
  {Relation.successful(Continuation.observe(machine)) == True{} : Bool}

# One complete actual production step preserves the meaning of all remaining work.
law step_preserves_continuation:
  for +machine: Core.Machine
  for valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool}
  {Continuation.observe(CoreTraversal.step(machine)) == Continuation.observe(machine) : Relation.Observation}

# At a terminal stack the entire accumulated flat plan is the recursive observation.
law terminal_observation_exact:
  for +machine: Core.Machine
  for terminal: {CoreTraversal.done(machine) == True{} : Bool}
  {
    Relation.observe_core(Core.CompletePlan{CoreTraversal.finish(machine)})
    == Continuation.observe(machine) : Relation.Observation
  }

# Adequate production fuel yields exactly the independent whole-stack result.
law run_observation_exact:
  for +fuel: Nat
  for +machine: Core.Machine
  for valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool}
  for enough: {Nat.is_le(Rank.rank(machine), fuel) == True{} : Bool}
  {
    Relation.observe_core(CoreTraversal.run(fuel, machine))
    == Continuation.observe(machine)
    : Relation.Observation
  }

# Production root entry corresponds to the complete independent recursive tree.
law initial_recursive_exact:
  for +facts: Core.Facts
  for +path: String
  for +name: String
  for +expected: Core.ExpectedKind
  for +visited: Map<&2, Bool>
  for +budget: Core.Budget
  for +depth: Nat
  for valid: {Patricia.canonical(Bool, visited) == True{} : Bool}
  {
    Continuation.initial(facts, path, name, expected, visited, budget, depth)
    == Relation.spec_observation(facts, path, name, expected, visited, budget, depth)
    : Relation.Observation
  }
