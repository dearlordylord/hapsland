# Whole nonempty-frame step. The different recursive request depths are exact;
# only the already-completed parents' adequate fuel is transported.
def nonempty_step_exact(
  +machine: Core.Machine,
  +reference: Core.Reference,
  +items: List<&2, Core.Reference>,
  +owner: Nat,
  +index: Nat,
  +depth: Nat,
  +fuel: Nat,
  +valid: {Patricia.canonical(Bool, CoreTraversal.machine_visited(machine)) == True{} : Bool},
  enough: {Nat.is_le(Relation.spec_bound(CoreTraversal.machine_facts(machine)), fuel) == True{} : Bool},
  head_enough: {
    Nat.is_le(
      Bound.cap(CoreTraversal.machine_facts(machine), List.length(&2, Core.Reference, reference <> items)),
      (1n + fuel : Nat)
    )
    == True{}
    : Bool
  },
  parent_before: {
    Bound.adequate(CoreTraversal.machine_facts(machine), (1n + fuel : Nat), State.frames(machine))
    == True{}
    : Bool
  },
  parent_after: {
    Bound.adequate(CoreTraversal.machine_facts(machine), fuel, State.frames(machine))
    == True{}
    : Bool
  }
) -> {
      Machine.evaluate(
        (1n + fuel : Nat),
        CoreTraversal.push_frame(machine, Core.Frame{owner, depth, index, reference <> items})
      )
      == Machine.evaluate(
        fuel,
        CoreTraversal.step(CoreTraversal.push_frame(machine, Core.Frame{owner, depth, index, reference <> items}))
      )
      : Relation.Observation
    }:
  match machine:
    case Core.Machine{
      +facts,
      +path,
      +parents,
      +visited,
      +budget,
      +address,
      +nodes,
      +slots,
      +pending
    }:
      +machine = {Core.Machine{facts, path, parents, visited, budget, address, nodes, slots, pending} : Core.Machine}
      %Head.head_exact(machine, reference, items, owner, index, depth, fuel, valid, enough):
        {
          Machine.evaluate(
            (1n + fuel : Nat),
            CoreTraversal.push_frame(machine, Core.Frame{owner, depth, index, reference <> items})
          )
          == _
          : Relation.Observation
        }
      Parent.first_frame_suffix(
        reference <> items,
        facts,
        path,
        owner,
        index,
        depth,
        Spec.State{visited, budget, address, pending},
        Prefix.prefix(machine),
        parents,
        (1n + fuel : Nat),
        (1n + fuel : Nat),
        fuel,
        valid,
        {==},
        head_enough,
        parent_before,
        parent_after
      )
