# One common adequate fuel for the complete before/after stacks.
def common(a: Nat, b: Nat, c: Nat) -> Nat:
  (a + (b + (c + 1n)) : Nat)

def first(+a: Nat, +b: Nat, +c: Nat) -> {Nat.is_le(a, common(a, b, c)) == True{} : Bool}:
  NatFacts.le_add_right(a, (b + (c + 1n) : Nat))

def second(+a: Nat, +b: Nat, +c: Nat) -> {Nat.is_le(b, common(a, b, c)) == True{} : Bool}:
  NatFacts.le_trans(b, (b + (c + 1n) : Nat), common(a, b, c),
    NatFacts.le_add_right(b, (c + 1n : Nat)), NatFacts.le_add_left((b + (c + 1n) : Nat), a))

def third(+a: Nat, +b: Nat, +c: Nat) -> {Nat.is_le(c, common(a, b, c)) == True{} : Bool}:
  NatFacts.le_trans(c, (c + 1n : Nat), common(a, b, c), NatFacts.le_add_right(c, 1n),
    NatFacts.le_trans((c + 1n : Nat), (b + (c + 1n) : Nat), common(a, b, c),
      NatFacts.le_add_left((c + 1n : Nat), b), NatFacts.le_add_left((b + (c + 1n) : Nat), a)))

def successor(
  +amount: Nat,
  +fuel: Nat,
  enough: {Nat.is_le(amount, fuel) == True{} : Bool}
) -> {Nat.is_le(amount, (1n + fuel : Nat)) == True{} : Bool}:
  NatFacts.le_trans(amount, fuel, (1n + fuel : Nat), enough, NatFacts.le_add_left(fuel, 1n))

def monotone(frames: List<&2, Core.Frame>, +facts: Core.Facts, +left: Nat, +right: Nat,
    +valid: {Bound.adequate(facts, left, frames) == True{} : Bool},
    +enough: {Nat.is_le(left, right) == True{} : Bool}) -> {
  Bound.adequate(facts, right, frames) == True{} : Bool
}:
  match frames:
    case []: {==}
    case Core.Frame{_, _, _, +items} <> +rest:
      +cap = Bound.cap(facts, List.length(&2, Core.Reference, items))
      Progress.conjoin(
        Nat.is_le(cap, right),
        Bound.adequate(facts, right, rest),
        NatFacts.le_trans(
          cap,
          left,
          right,
          Order.eq_true_of_and_left(Nat.is_le(cap, left), Bound.adequate(facts, left, rest), valid),
          enough
        ),
        monotone(
          rest,
          facts,
          left,
          right,
          Order.eq_true_of_and_right(
            Nat.is_le(cap, left),
            Bound.adequate(facts, left, rest),
            valid
          ),
          enough
        )
      )

def positive(+a: Nat, +b: Nat, +c: Nat) -> {Nat.is_le(1n, common(a, b, c)) == True{} : Bool}:
  NatFacts.le_trans(1n, (c + 1n : Nat), common(a, b, c), NatFacts.le_add_left(1n, c),
    NatFacts.le_trans((c + 1n : Nat), (b + (c + 1n) : Nat), common(a, b, c),
      NatFacts.le_add_left((c + 1n : Nat), b), NatFacts.le_add_left((b + (c + 1n) : Nat), a)))
