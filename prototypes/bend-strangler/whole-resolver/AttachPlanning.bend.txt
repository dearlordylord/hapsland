def work(graph: Graph.Graph) -> Nat:
  match graph:
    case Graph.Graph{_, _, _, _, _, _, work, _, _, _, _}: work

def local_work(budget: Local.Budget) -> Nat:
  match budget:
    case Local.Budget{_, _, _, work, _, _}: work

def with_graph_work(budget: Local.Budget, value: Nat) -> Local.Budget:
  match budget:
    case Local.Budget{limits, targets, max_targets, work, _, depth}:
      Local.Budget{limits, targets, max_targets, work, value, depth}

def plan_budget(plan: Local.Plan) -> Local.Budget:
  match plan:
    case Local.Plan{_, _, _, _, _, budget}: budget

def plan_visited(plan: Local.Plan) -> Map<&2, Bool>:
  match plan:
    case Local.Plan{_, _, _, _, visited, _}: visited

def target_id(id: Maybe<&2, Nat>, frame: T.ResolverFrame) -> Maybe<&2, T.Target>:
  match id:
    case None{}: None{}
    case Some{id}: Local.map_result(
      Maybe<&2, T.Target>,
      Map.get(Maybe<&2, T.Target>, None{}, S.frame_target_paths(frame), Nat.show(id))
    )

def target(state: T.ResolverState, frame: T.ResolverFrame) -> Maybe<&2, T.Target>:
  target_id(S.scratch_active_target(S.state_scratch(state)), frame)

def pending(target: T.Target) -> Local.Pending:
  match target:
    case T.FileTarget{_, _, edge}: edge
    case T.BundledTarget{_, _, edge}: edge

def optional_number(id: Maybe<&2, Nat>) -> Nat:
  match id:
    case None{}: 0n
    case Some{id}: id

def target_number(state: T.ResolverState) -> Nat:
  optional_number(S.scratch_active_target(S.state_scratch(state)))

def optional_capture(capture: Maybe<&2, T.CaptureToken>, state: T.ResolverState) -> T.CaptureToken:
  match capture:
    case None{}: S.input_root_capture(S.state_input(state))
    case Some{capture}: capture

def child_capture(+state: T.ResolverState) -> T.CaptureToken:
  optional_capture(S.scratch_capture(S.state_scratch(state)), state)

def request_previous(
  product: Maybe<&2, T.ProductValue>,
  state: T.ResolverState
) -> Loop.Continuation:
  match product:
    case None{}: ResolveOwner.malformed()
    case Some{value}: Loop.ServiceStep{
      S.request(state, T.MeasuringPreviousTree{}, T.EncodeProduct{value})
    }

def measure_child(
  bundled: Bool,
  +state: T.ResolverState,
  frame: T.ResolverFrame
) -> Loop.Continuation:
  match bundled:
    case True{}: Measurement.previous(S.scratch_bundled_previous(S.state_scratch(state)), state)
    case False{}: request_previous(Root.unit(frame), state)

def built_allowed(
  allowed: Bool,
  +plan: Local.Plan,
  file: T.CapturedFacts,
  +declaration: Local.Declaration,
  +state: T.ResolverState,
  +frame: T.ResolverFrame
) -> Loop.Continuation:
  match allowed:
    case False{}: Read.failed(
      S.with_state_frame(state, Some{S.with_frame_budget(frame, plan_budget(plan))})
    )
    case True{}:
      match declaration:
        case Local.Declaration{handle, identity, _, _, _}:
          child : T.TentativeChild = T.TentativeChild{
            target_number(state),
            S.scratch_capture_path(S.state_scratch(state)),
            child_capture(state),
            S.scratch_source_bytes(S.state_scratch(state)),
            file,
            handle,
            identity,
            local_work(S.frame_budget(frame)),
            plan,
            0n,
            T.ProductNull{},
            []
          }
          measure_child(Measurement.bundled_state(state), S.with_state_frame(S.with_state_scratch(state,
            S.with_scratch_tentative(S.state_scratch(state), Some{child})),
            Some{S.with_frame_budget(frame, plan_budget(plan))}), frame)

def global_allowed(
  allowed: Bool,
  +plan: Local.Plan,
  file: T.CapturedFacts,
  declaration: Local.Declaration,
  +state: T.ResolverState,
  +frame: T.ResolverFrame
) -> Loop.Continuation:
  match allowed:
    case False{}: Resolve.progressed(
      False{},
      state,
      S.advance(
        S.with_frame_budget(frame, plan_budget(plan)),
        Graph.Captured{
          S.scratch_source_bytes(S.state_scratch(state)),
          0n,
          Nat.sub(local_work(plan_budget(plan)), local_work(S.frame_budget(frame))),
          []
        }
      )
    )
    case True{}: built_allowed(
      RootOwner.permits(S.frame_limits(frame), plan_budget(plan)),
      plan,
      file,
      declaration,
      state,
      frame
    )

def plan_allowed(budget: Local.Budget, limits: Graph.Limits) -> Bool:
  match budget:
    case Local.Budget{_, _, _, local_work, graph_work, _}: Graph.local_budget(
      limits,
      local_work,
      0n,
      0n,
      graph_work
    )

def planned_complete(
  bundled: Bool,
  +plan: Local.Plan,
  file: T.CapturedFacts,
  declaration: Local.Declaration,
  +state: T.ResolverState,
  +frame: T.ResolverFrame
) -> Loop.Continuation:
  match bundled:
    case True{}: built_allowed(
      RootOwner.permits(S.frame_limits(frame), plan_budget(plan)),
      plan,
      file,
      declaration,
      state,
      frame
    )
    case False{}: global_allowed(
      plan_allowed(plan_budget(plan), S.frame_limits(frame)),
      plan,
      file,
      declaration,
      state,
      frame
    )

def planned(result: Local.RunResult, file: T.CapturedFacts, declaration: Local.Declaration,
    +state: T.ResolverState, +frame: T.ResolverFrame) -> Loop.Continuation:
  match result:
    case Local.InvariantFailure{_}: Loop.ServiceStep{T.FailedResolver{T.LocalTraversalFailure{}}}
    case Local.MissingRoot{_, budget}: Read.failed(
      S.with_state_frame(state, Some{S.with_frame_budget(frame, budget)})
    )
    case Local.CompletePlan{+plan}:
      planned_complete(Measurement.bundled_state(state), plan, file, declaration, state, frame)

def empty_or_plan(
  seen: Bool,
  +facts: Local.Facts,
  path: String,
  edge: Local.Pending,
  +declaration: Local.Declaration,
  visited: Map<&2, Bool>,
  budget: Local.Budget
) -> Local.RunResult:
  match seen:
    case True{}:
      match declaration:
        case Local.Declaration{handle, _, _, _, _}: Local.CompletePlan{
          Local.Plan{0n, [Local.PlannedNode{0n, handle}], [], [], visited, budget}}
    case False{}:
      match edge:
        case Local.Pending{_, _, _, _, name, depth, expected, _}:
          LocalTraversal.plan(facts, path, name, expected, visited, budget, (depth + 1n : Nat))

def begin_plan(negative: Bool, +facts: Local.Facts, exported: Map<&2, Bool>,
    edge: Local.Pending, +declaration: Local.Declaration, +state: T.ResolverState,
    +frame: T.ResolverFrame) -> Loop.Continuation:
  match negative:
    case True{}: Loop.ServiceStep{T.FailedResolver{T.NegativeGraphWork{
      Nat.sub(local_work(S.frame_budget(frame)), work(S.frame_graph(frame)))}}}
    case False{}:
      +budget = with_graph_work(
        S.frame_budget(frame),
        Nat.sub(work(S.frame_graph(frame)), local_work(S.frame_budget(frame)))
      )
      +identity = LocalTraversal.declaration_identity(declaration)
      planned(
        empty_or_plan(
          Bool.and(
            Bool.not(Measurement.bundled_state(state)),
            Local.map_result(Bool, Map.get(Bool, False{}, S.frame_visited(frame), identity))
          ),
          facts,
          S.scratch_capture_path(S.state_scratch(state)),
          edge,
          declaration,
          S.frame_visited(frame),
          budget
        ),
        T.CapturedFacts{facts, exported, S.scratch_source_bytes(S.state_scratch(state))},
        declaration,
        state,
        frame
      )

def chosen_declaration(
  declaration: Maybe<&2, Local.Declaration>,
  +facts: Local.Facts,
  +exported: Map<&2, Bool>,
  +edge: Local.Pending,
  +state: T.ResolverState,
  +frame: T.ResolverFrame
) -> Loop.Continuation:
  match declaration:
    case None{}: Read.failed(state)
    case Some{declaration}:
      yes : Unit -> Loop.Continuation = _ => begin_plan(
        Nat.is_gt(local_work(S.frame_budget(frame)), work(S.frame_graph(frame))),
        facts,
        exported,
        edge,
        declaration,
        state,
        frame
      )
      no : Unit -> Loop.Continuation = _ => Read.failed(state)
      Materialize.decide(Loop.Continuation, Resolve.supported_edge(edge, facts, exported), yes, no)

def edge_expected(edge: Local.Pending) -> Local.ExpectedKind:
  match edge:
    case Local.Pending{_, _, _, _, _, _, expected, _}: expected

def prepared(source: Maybe<&2, Syntax.PreparedFacts>, target: Maybe<&2, T.Target>,
    +state: T.ResolverState, +frame: T.ResolverFrame) -> Loop.Continuation:
  match source target:
    case Some{Syntax.PreparedFacts{+facts, exported, next, artifacts}} Some{
      T.FileTarget{_, _, +edge}
    }:
      +updated = S.with_frame_next_artifact(S.with_frame_artifacts(frame,
        List.append(&2, T.ArtifactEntry, S.frame_artifacts(frame), artifacts)), next)
      chosen_declaration(
        LocalTraversal.declaration_lookup(facts, Resolve.pending_name(edge), edge_expected(edge)),
        facts,
        exported,
        edge,
        S.with_state_frame(
          S.with_state_scratch(
            state,
            S.with_scratch_pending_edge(S.state_scratch(state), Some{edge})
          ),
          Some{updated}
        ),
        updated
      )
    case _ _: ResolveOwner.malformed()

def inspected_frame(
  frame: Maybe<&2, T.ResolverFrame>,
  facts: T.SourceFacts,
  +state: T.ResolverState
) -> Loop.Continuation:
  match frame:
    case None{}: ResolveOwner.malformed()
    case Some{+frame}: prepared(
      Syntax.prepare(facts, S.frame_next_artifact(frame)),
      target(state, frame),
      state,
      frame
    )

def inspected(+state: T.ResolverState, facts: T.SourceFacts) -> Loop.Continuation:
  inspected_frame(S.state_frame(state), facts, state)

def published(+frame: T.ResolverFrame, target: Nat, +path: String, file: T.CapturedFacts,
    identity: String, plan: Local.Plan) -> T.ResolverFrame:
  S.with_frame_target_artifacts(
    S.with_frame_captured(
      S.with_frame_visited(frame, plan_visited(plan)),
      Map.set(&2, Maybe<&2, T.CapturedFacts>, S.frame_captured(frame), path, Some{file})
    ),
    Map.set(
      &2,
      Maybe<&2, String>,
      S.frame_target_artifacts(frame),
      Nat.show(target),
      Some{identity}
    )
  )

def rejected_slots(slots: Maybe<&2, List<&2, Local.ReferenceSlot>>, frame: T.ResolverFrame,
    state: T.ResolverState) -> Loop.Continuation:
  match slots:
    case None{}: ResolveOwner.malformed()
    case Some{slots}: Resolve.progressed(False{}, state, S.with_frame_references(frame, slots))

def publish_child(bundled: Bool, +frame: T.ResolverFrame, target: Nat, path: String,
    file: T.CapturedFacts, identity: String, plan: Local.Plan) -> T.ResolverFrame:
  match bundled:
    case False{}: published(frame, target, path, file, identity, plan)
    case True{}: S.with_frame_target_artifacts(
      S.with_frame_visited(frame, plan_visited(plan)),
      Map.set(
        &2,
        Maybe<&2, String>,
        S.frame_target_artifacts(frame),
        Nat.show(target),
        Some{identity}
      )
    )

def rejection_name(edge: Local.Pending) -> String:
  match edge:
    case Local.Pending{_, _, _, symbol, name, _, _, target}: Bool.pick(
      String,
      Measurement.bundled_target(target),
      symbol,
      name
    )

def admission(command: Graph.Command, child: T.TentativeChild, +edge: Local.Pending,
    +frame: T.ResolverFrame, state: T.ResolverState) -> Loop.Continuation:
  match command:
    case Graph.SkipImport{_, _}: rejected_slots(
      Compose.replacement(
        Compose.replace_slot(
          S.frame_references(frame),
          Measurement.edge_owner(edge),
          Measurement.edge_index(edge),
          Local.Omitted{Local.ReferenceLimit{}, rejection_name(edge)}
        )
      ),
      frame,
      state
    )
    case _:
      match child:
        case T.TentativeChild{target, path, _, _, file, _, identity, _, plan, _, _, _}:
          Resolve.progressed(
            False{},
            state,
            publish_child(Measurement.bundled_edge(edge), frame, target, path, file, identity, plan)
          )

def candidate_child(+child: T.TentativeChild, +frame: T.ResolverFrame,
    edge: Local.Pending, bytes: Nat, state: T.ResolverState) -> Loop.Continuation:
  match child:
    case T.TentativeChild{_, _, _, source_bytes, _, _, _, before, _, previous, _, ids}:
      +updated = S.advance(frame, Graph.Captured{source_bytes, Nat.sub(bytes, previous),
        Nat.sub(local_work(S.frame_budget(frame)), before), ids})
      admission(S.frame_command(updated), child, edge, updated, state)

def candidate_parts(child: Maybe<&2, T.TentativeChild>, frame: Maybe<&2, T.ResolverFrame>,
    edge: Maybe<&2, Local.Pending>, bytes: Nat, state: T.ResolverState) -> Loop.Continuation:
  match child frame edge:
    case Some{child} Some{frame} Some{edge}: candidate_child(child, frame, edge, bytes, state)
    case _ _ _: ResolveOwner.malformed()

def candidate(bytes: Nat, +state: T.ResolverState) -> Loop.Continuation:
  candidate_parts(S.scratch_tentative(S.state_scratch(state)), S.state_frame(state),
    S.scratch_pending_edge(S.state_scratch(state)), bytes, state)

def declaration_handle(
  items: List<&2, Maybe<&2, Local.Declaration>>,
  +handle: Nat
) -> Maybe<&2, Local.Declaration>:
  match items:
    case []: None{}
    case None{} <> tail: declaration_handle(tail, handle)
    case Some{+declaration} <> tail:
      match declaration:
        case Local.Declaration{actual, _, _, _, _}:
          yes : Unit -> Maybe<&2, Local.Declaration> = _ => Some{declaration}
          no : Unit -> Maybe<&2, Local.Declaration> = _ => declaration_handle(tail, handle)
          Materialize.decide(Maybe<&2, Local.Declaration>, Nat.is_eq(actual, handle), yes, no)

def supporting_handle(facts: Local.Facts, handle: Nat) -> Maybe<&2, Local.Declaration>:
  match facts:
    case Local.Facts{_, _, _, supporting}: declaration_handle(
      Map.values(&2, Maybe<&2, Local.Declaration>, supporting),
      handle
    )

def artifact_name(artifact: T.Artifact) -> String:
  match artifact:
    case T.Artifact{_, _, _, _, name, _, _, _}: name

def artifact_source(artifact: T.Artifact) -> String:
  match artifact:
    case T.Artifact{_, _, _, _, _, source, _, _}: source

def bundle_facts(facts: Local.Facts, name: String, declaration: Local.Declaration) -> Local.Facts:
  match facts:
    case Local.Facts{aware, _, imports, supporting}: Local.Facts{
      aware,
      Map.set(&2, Maybe<&2, Local.Declaration>, MTip{}, name, Some{declaration}),
      imports,
      supporting
    }

def bundle_plan_edge(edge: Local.Pending, name: String) -> Local.Pending:
  match edge:
    case Local.Pending{owner, index, from, symbol, _, depth, _, target}:
      Local.Pending{owner, index, from, symbol, name, depth, Local.AnyKind{}, target}

def bundle_lookup(found: Maybe<&2, Local.Declaration>, +facts: Local.Facts,
    edge: Local.Pending, +state: T.ResolverState, +frame: T.ResolverFrame) -> Loop.Continuation:
  match found:
    case None{}: Read.failed(state)
    case Some{declaration}: begin_plan(
      Nat.is_gt(local_work(S.frame_budget(frame)), work(S.frame_graph(frame))),
      facts, MTip{}, edge, declaration, state, frame)

def bundle_ready(
  artifact: Maybe<&2, T.Artifact>,
  declaration: Maybe<&2, Local.Declaration>,
  facts: Local.Facts,
  +edge: Local.Pending,
  +state: T.ResolverState,
  frame: T.ResolverFrame
) -> Loop.Continuation:
  match artifact declaration:
    case Some{+artifact} Some{declaration}:
      +file = bundle_facts(facts, artifact_name(artifact), declaration)
      bundle_lookup(
        LocalTraversal.declaration_lookup(file, artifact_name(artifact), Local.AnyKind{}),
        file,
        bundle_plan_edge(edge, artifact_name(artifact)),
        S.with_state_scratch(
          state,
          S.with_scratch_capture_path(S.state_scratch(state), Resolve.from_path(edge))
        ),
        frame
      )
    case _ _: ResolveOwner.malformed()

def bundle_previous_product(
  product: Maybe<&2, T.ProductValue>,
  state: T.ResolverState
) -> Loop.Continuation:
  match product:
    case None{}: ResolveOwner.malformed()
    case Some{value}: Loop.ServiceStep{
      S.request(state, T.MeasuringBundledPrevious{}, T.EncodeProduct{value})
    }

def bundle_bytes_parts(
  frame: Maybe<&2, T.ResolverFrame>,
  target: Maybe<&2, T.Target>,
  +bytes: Nat,
  +state: T.ResolverState
) -> Loop.Continuation:
  match frame target:
    case Some{+frame} Some{T.BundledTarget{+handle, +facts, +edge}}:
      yes : Unit -> Loop.Continuation = _ => bundle_ready(
        Resolve.artifact_at(S.frame_artifacts(frame), handle),
        supporting_handle(facts, handle),
        facts,
        edge,
        S.with_state_scratch(
          state,
          S.with_scratch_source_bytes(
            S.with_scratch_pending_edge(S.state_scratch(state), Some{edge}),
            bytes
          )
        ),
        frame
      )
      no : Unit -> Loop.Continuation = _ => Resolve.progressed(
        False{},
        state,
        S.advance(frame, Graph.Captured{bytes, 0n, 0n, []})
      )
      Materialize.decide(
        Loop.Continuation,
        Nat.is_le(bytes, Graph.source_cap(S.frame_limits(frame))),
        yes,
        no
      )
    case _ _: ResolveOwner.malformed()

def bundle_bytes_frame(
  frame: Maybe<&2, T.ResolverFrame>,
  bytes: Nat,
  +state: T.ResolverState
) -> Loop.Continuation:
  match frame:
    case None{}: ResolveOwner.malformed()
    case Some{+frame}: bundle_bytes_parts(Some{frame}, target(state, frame), bytes, state)

def bundle_bytes(bytes: Nat, +state: T.ResolverState) -> Loop.Continuation:
  bundle_bytes_frame(S.state_frame(state), bytes, state)

def bundle_source(artifact: Maybe<&2, T.Artifact>, state: T.ResolverState) -> Loop.Continuation:
  match artifact:
    case None{}: ResolveOwner.malformed()
    case Some{artifact}: Loop.ServiceStep{
      S.request(state, T.MeasuringBundledSource{}, T.EncodeSourceText{artifact_source(artifact)})
    }

def bundle_frame(
  frame: Maybe<&2, T.ResolverFrame>,
  handle: Nat,
  state: T.ResolverState
) -> Loop.Continuation:
  match frame:
    case None{}: ResolveOwner.malformed()
    case Some{frame}: bundle_source(Resolve.artifact_at(S.frame_artifacts(frame), handle), state)

def bundled(+state: T.ResolverState, handle: Nat) -> Loop.Continuation:
  bundle_frame(S.state_frame(state), handle, state)

def bundle_previous_frame(
  frame: Maybe<&2, T.ResolverFrame>,
  target: Maybe<&2, T.Target>,
  state: T.ResolverState
) -> Loop.Continuation:
  match frame target:
    case Some{+frame} Some{T.BundledTarget{+handle, +facts, edge}}: bundle_ready(
      Resolve.artifact_at(S.frame_artifacts(frame), handle),
      supporting_handle(facts, handle),
      facts,
      edge,
      state,
      frame
    )
    case _ _: ResolveOwner.malformed()

def bundle_previous_state(
  frame: Maybe<&2, T.ResolverFrame>,
  +state: T.ResolverState
) -> Loop.Continuation:
  match frame:
    case None{}: ResolveOwner.malformed()
    case Some{+frame}: bundle_previous_frame(Some{frame}, target(state, frame), state)

def bundle_previous(bytes: Nat, +state: T.ResolverState) -> Loop.Continuation:
  bundle_previous_state(S.state_frame(state), S.with_state_scratch(state,
    S.with_scratch_bundled_previous(S.state_scratch(state), bytes)))

def reply(
  stage: T.ResolverStage,
  outcome: T.ServiceOutcome,
  state: T.ResolverState
) -> Loop.Continuation:
  match stage outcome:
    case T.MeasuringBundledPrevious{} T.ProductEncoded{bytes}: bundle_previous(bytes, state)
    case T.MeasuringBundledSource{} T.SourceTextEncoded{bytes}: bundle_bytes(bytes, state)
    case T.MeasuringPreviousTree{} T.ProductEncoded{bytes}: Measurement.previous(bytes, state)
    case T.MeasuringCandidateTree{} T.ProductEncoded{bytes}: candidate(bytes, state)
    case _ _: Read.reply(stage, outcome, state)


def main() -> Unit:
  Unit{}
