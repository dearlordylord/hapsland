law successful_flatten_preserves_construction_allowance:
  for +fuel: Nat
  for +request: Recursive.FlattenRequest
  for episode: Nat
  for route: List<&2, Nat>
  for artifacts: List<&2, T.ArtifactEntry>
  for adequate: {C.flatten_success(Recursive.flatten(fuel, request)) == True{} : Bool}
  {
    C.not_exhausted(Forest.construct_interpret(fuel,
      C.construction_request(request, episode, route, artifacts))) == True{}
    : Bool
  }
