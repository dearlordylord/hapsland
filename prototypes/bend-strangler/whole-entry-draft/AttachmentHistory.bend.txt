# Fixed measurement history extension from the common ready phase. It does not
# inspect an actual/ref result or choose a witness after observing the output.
def ids(items: List<&2, V.PendingPayload>, +next: Nat) -> List<&2, Nat>:
  match items:
    case []: []
    case _ <> tail: next <> ids(tail, (next + 1n : Nat))

def provenance(items: List<&2, Nat>, +source: T.SourceFacts,
    +catalog: Root.Source) -> List<&2, V.EdgeSource>:
  match items:
    case []: []
    case head <> tail: V.EdgeSource{head, source, catalog} <> provenance(tail, source, catalog)

def extend(history: V.History, child_addresses: List<&2, F.AddressEntry>,
    offset: Nat, +new_ids: List<&2, Nat>, source: T.SourceFacts,
    catalog: Root.Source) -> V.History:
  match history:
    case V.History{addresses, V.Order{pending, targets, ids, captured, identities, visited},
      edges, captures, artifacts, remaining, episode}:
      V.History{
        List.append(&2, F.AddressEntry, addresses, Mapping.shifted(child_addresses, offset)),
        V.Order{
          List.append(&2, Nat, pending, new_ids),
          targets,
          ids,
          captured,
          identities,
          visited
        },
        List.append(&2, V.EdgeSource, edges, provenance(new_ids, source, catalog)),
        captures,
        artifacts,
        remaining,
        (episode + 1n : Nat)
      }

def measurement(phase: Phase.Phase, history: V.History) -> V.History:
  match phase:
    case Phase.Phase{
      V.Frame{_, _, _, _, _, _, _, _, _, _, _, _, _, node, edge, _, _, _, _, _},
      Phase.Child{_, _, _, _, _, _, _, _, _, _, addresses, pending, _, _, source, catalog}, _}:
      extend(history, addresses, node, ids(pending, edge), source, catalog)

def has_nat(items: List<&2, Nat>, +value: Nat) -> Bool:
  match items:
    case []: False{}
    case head <> tail: Bool.or(Nat.is_eq(head, value), has_nat(tail, value))

def has_string(items: List<&2, String>, +value: String) -> Bool:
  match items:
    case []: False{}
    case head <> tail: Bool.or(String.eq(head, value), has_string(tail, value))

def retain_nat(+items: List<&2, Nat>, +value: Nat) -> List<&2, Nat>:
  Bool.pick(List<&2, Nat>, has_nat(items, value), items, List.append(&2, Nat, items, [value]))

def retain_string(+items: List<&2, String>, +value: String) -> List<&2, String>:
  Bool.pick(
    List<&2, String>,
    has_string(items, value),
    items,
    List.append(&2, String, items, [value])
  )

def store_capture(items: List<&2, V.CaptureSource>, +path: String,
    +source: T.SourceFacts, +catalog: Root.Source) -> List<&2, V.CaptureSource>:
  match items:
    case []: [V.CaptureSource{path, source, catalog}]
    case +head <> +tail:
      match head:
        case V.CaptureSource{key, _, _}:
          Bool.pick(List<&2, V.CaptureSource>, String.eq(path, key),
            V.CaptureSource{path, source, catalog} <> tail,
            head <> store_capture(tail, path, source, catalog))

def published(
  history: V.History,
  kind: X.ChildKind,
  target: Nat,
  +path: String,
  source: T.SourceFacts,
  catalog: Root.Source,
  visited: List<&2, String>,
  remaining: Nat
) -> V.History:
  match history kind:
    case V.History{addresses, V.Order{pending, targets, ids, captured, identities, _},
      edges, captures, artifacts, _, episode} X.FileChild{_}:
      V.History{addresses, V.Order{pending, targets, ids, retain_string(captured, path),
        retain_nat(identities, target), visited}, edges,
        store_capture(captures, path, source, catalog), artifacts, remaining, episode}
    case V.History{addresses, V.Order{pending, targets, ids, captured, identities, _},
      edges, captures, artifacts, _, episode} X.BundledChild{}:
      V.History{addresses, V.Order{pending, targets, ids, captured,
        retain_nat(identities, target), visited}, edges, captures, artifacts, remaining, episode}

def allowance(history: V.History, remaining: Nat) -> V.History:
  match history:
    case V.History{addresses, order, edges, captures, artifacts, _, episode}:
      V.History{addresses, order, edges, captures, artifacts, remaining, episode}

def after_graph(result: G.BoundedStep, child: Phase.Child, +history: V.History) -> V.History:
  match result child:
    case G.BoundedStep{G.Bounded{_, remaining}, G.SkipImport{_, _}} _:
      allowance(history, remaining)
    case G.BoundedStep{G.Bounded{_, remaining}, _}
      Phase.Child{target, path, kind, _, _, _, _, _, _, _, _, _, visited, _, source, catalog}:
      published(history, kind, target, path, source, catalog, visited, remaining)

def encoded_child(+child: Phase.Child, history: V.History, graph: G.Graph,
    remaining: Nat, next_edge: Nat, previous: Nat, bytes: Nat) -> V.History:
  match child:
    case Phase.Child{_, _, _, _, source_bytes, _, _, _, before, _, _, pending, _,
      V.Budget{_, _, _, work, _, _}, _, _}:
      after_graph(
        Policy.bounded_transition(
          G.Bounded{graph, remaining},
          G.Captured{
            source_bytes,
            Nat.sub(bytes, previous),
            Nat.sub(work, before),
            ids(pending, next_edge)
          }
        ),
        child,
        history
      )

def encoded(phase: Phase.Phase, +history: V.History, previous: Nat, bytes: Nat) -> V.History:
  match phase history:
    case Phase.Phase{
      V.Frame{_, _, graph, _, _, _, _, _, _, _, _, _, _, _, next_edge, _, _, _, _, _},
      child, _} V.History{_, _, _, _, _, remaining, _}:
      encoded_child(child, history, graph, remaining, next_edge, previous, bytes)

def accepted(validation: Maybe<&2, T.TechnicalFailure>, phase: Phase.Phase,
    history: V.History, previous: Nat, reply: T.Reply) -> V.History:
  match validation reply:
    case None{} T.Reply{_, _, T.ProductEncoded{bytes}}: encoded(phase, history, previous, bytes)
    case _ _: history

# Request is the common EncodeProduct request established by the entry
# conclusion, not an outcome-dependent callback or supplied expected history.
def reply(phase: Phase.Phase, +history: V.History, previous: Nat,
    +request: T.Request, +response: T.Reply) -> V.History:
  accepted(Protocol.check(request, response), phase, history, previous, response)
