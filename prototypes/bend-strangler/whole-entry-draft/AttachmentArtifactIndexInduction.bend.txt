# Internal induction invariant for the one full existing index fold.
def distinct(entries: List<&2, T.ArtifactEntry>) -> Bool:
  Model.distinct(entries)

def absent(entries: List<&2, T.ArtifactEntry>, +index: Map<&2, Maybe<&2, T.Artifact>>) -> Bool:
  Model.absent(entries, index)

def expected(entries: List<&2, T.ArtifactEntry>, +handle: Nat, prior: Maybe<&2, T.Artifact>) -> Maybe<&2, T.Artifact>:
  Model.expected(entries, handle, prior)

def accumulated(+entries: List<&2, T.ArtifactEntry>, +prior: Map<&2, Maybe<&2, T.Artifact>>,
    +result: Map<&2, Maybe<&2, T.Artifact>>) -> Type:
  Model.accumulated(entries, prior, result)

def read_rows(entries: List<&2, T.ArtifactEntry>, +written: Nat, +artifact: T.Artifact,
    +index: Map<&2, Maybe<&2, T.Artifact>>) -> Type:
  match entries:
    case []: Unit
    case T.ArtifactEntry{+id, _} <> tail:
      {Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(id))
      == Bool.pick(Maybe<&2, T.Artifact>, String.eq(Nat.show(id), Nat.show(written)), Some{artifact}, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(id))) : Maybe<&2, T.Artifact>} & read_rows(tail, written, artifact, index)

def other_read(+wanted: Nat, +written: Nat, +artifact: T.Artifact, +index: Map<&2, Maybe<&2, T.Artifact>>,
    head_read: {Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(wanted))
      == Bool.pick(Maybe<&2, T.Artifact>, String.eq(Nat.show(wanted), Nat.show(written)), Some{artifact}, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(wanted))) : Maybe<&2, T.Artifact>},
    different: {Nat.is_eq(written, wanted) == False{} : Bool},
    valid: {Patricia.canonical(Maybe<&2, T.Artifact>, index) == True{} : Bool}) ->
    {Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(wanted))
      == Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(wanted)) : Maybe<&2, T.Artifact>}:
  key_different: {String.eq(Nat.show(wanted), Nat.show(written)) == False{} : Bool} =
    Equal.trans(Bool, String.eq(Nat.show(wanted), Nat.show(written)), Nat.is_eq(wanted, written), False{},
      Keys.natural_key_equality(wanted, written),
      Equal.trans(Bool, Nat.is_eq(wanted, written), Nat.is_eq(written, wanted), False{}, N.is_eq_comm(wanted, written), different))
  %key_different:
    {Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(wanted))
      == Bool.pick(Maybe<&2, T.Artifact>, _, Some{artifact}, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(wanted))) : Maybe<&2, T.Artifact>}
  head_read

def none_of_fresh(+value: Maybe<&2, T.Artifact>, fresh: {Bool.not(Maybe.is_some(&2, T.Artifact, value)) == True{} : Bool}) ->
    {value == None{} : Maybe<&2, T.Artifact>}:
  match value:
    case None{}: {==}
    case Some{artifact}: Empty.absurd({Some{artifact} == None{} : Maybe<&2, T.Artifact>}, B.internal_false_ne_true(fresh))

def absent_finish(+id: Nat, +tail: List<&2, T.ArtifactEntry>, +written: Nat, +artifact: T.Artifact,
    +index: Map<&2, Maybe<&2, T.Artifact>>,
    head_read: {Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(id))
      == Bool.pick(Maybe<&2, T.Artifact>, String.eq(Nat.show(id), Nat.show(written)), Some{artifact}, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(id))) : Maybe<&2, T.Artifact>},
    parts: {Nat.is_eq(written, id) == False{} : Bool} & {P.handle_count(tail, written) == 0n : Nat},
    +fresh: {Bool.and(Bool.not(Maybe.is_some(&2, T.Artifact, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(id)))), absent(tail, index)) == True{} : Bool},
    valid: {Patricia.canonical(Maybe<&2, T.Artifact>, index) == True{} : Bool},
    recursive: {P.handle_count(tail, written) == 0n : Nat} -> {absent(tail, index) == True{} : Bool} ->
      {absent(tail, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact})) == True{} : Bool}) ->
    {Bool.and(Bool.not(Maybe.is_some(&2, T.Artifact,
      Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(id)))),
      absent(tail, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}))) == True{} : Bool}:
  match parts:
    case (different, rest_zero):
      Distinct.and_true(Bool.not(Maybe.is_some(&2, T.Artifact,
          Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(id)))),
        absent(tail, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact})),
        %Equal.sym(Maybe<&2, T.Artifact>,
          Core.map_read(Maybe<&2, T.Artifact>, None{}, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact}), Nat.show(id)),
          Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(id)), other_read(id, written, artifact, index, head_read, different, valid)):
          {Bool.not(Maybe.is_some(&2, T.Artifact, _)) == True{} : Bool}
        B.eq_true_of_and_left(Bool.not(Maybe.is_some(&2, T.Artifact, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(id)))), absent(tail, index), fresh),
        recursive(rest_zero)(B.eq_true_of_and_right(Bool.not(Maybe.is_some(&2, T.Artifact, Core.map_read(Maybe<&2, T.Artifact>, None{}, index, Nat.show(id)))), absent(tail, index), fresh)))

def absent_after_set(entries: List<&2, T.ArtifactEntry>, +written: Nat, +artifact: T.Artifact,
    +index: Map<&2, Maybe<&2, T.Artifact>>,
    rows: read_rows(entries, written, artifact, index),
    zero: {P.handle_count(entries, written) == 0n : Nat},
    fresh: {absent(entries, index) == True{} : Bool},
    +valid: {Patricia.canonical(Maybe<&2, T.Artifact>, index) == True{} : Bool}) ->
    {absent(entries, Map.set(&2, Maybe<&2, T.Artifact>, index, Nat.show(written), Some{artifact})) == True{} : Bool}:
  match entries rows:
    case [] Unit{}: {==}
    case T.ArtifactEntry{+id, _} <> +tail Tuple{head_read, tail_reads}:
      absent_finish(id, tail, written, artifact, index, head_read,
        Distinct.zero_parts(Nat.is_eq(written, id), P.handle_count(tail, written), zero), fresh, valid,
        rest_zero => tail_fresh => absent_after_set(tail, written, artifact, index, tail_reads, rest_zero, tail_fresh, valid))

def expected_zero_finish(+id: Nat, +artifact: T.Artifact, +tail: List<&2, T.ArtifactEntry>, +handle: Nat,
    +prior: Maybe<&2, T.Artifact>,
    parts: {Nat.is_eq(handle, id) == False{} : Bool} & {P.handle_count(tail, handle) == 0n : Nat},
    recursive: {P.handle_count(tail, handle) == 0n : Nat} -> {expected(tail, handle, prior) == prior : Maybe<&2, T.Artifact>}) ->
    {expected(T.ArtifactEntry{id, artifact} <> tail, handle, prior) == prior : Maybe<&2, T.Artifact>}:
  match parts:
    case Tuple{different, rest_zero}:
      opposite: {Nat.is_eq(id, handle) == False{} : Bool} =
        Equal.trans(Bool, Nat.is_eq(id, handle), Nat.is_eq(handle, id), False{}, N.is_eq_comm(id, handle), different)
      %Equal.sym(Bool, Nat.is_eq(id, handle), False{}, opposite):
        {Bool.pick(Maybe<&2, T.Artifact>, _, Some{artifact}, expected(tail, handle, prior)) == prior : Maybe<&2, T.Artifact>}
      recursive(rest_zero)

def expected_zero(entries: List<&2, T.ArtifactEntry>, +handle: Nat, +prior: Maybe<&2, T.Artifact>,
    zero: {P.handle_count(entries, handle) == 0n : Nat}) ->
    {expected(entries, handle, prior) == prior : Maybe<&2, T.Artifact>}:
  match entries:
    case []: {==}
    case T.ArtifactEntry{+id, +artifact} <> +tail:
      expected_zero_finish(id, artifact, tail, handle, prior,
        Distinct.zero_parts(Nat.is_eq(handle, id), P.handle_count(tail, handle), zero),
        rest_zero => expected_zero(tail, handle, prior, rest_zero))
